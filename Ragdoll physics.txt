local function FindShooterDebug()
    local function tryGet(parent)
        if not parent then return nil end
        local ok, result = pcall(function() return parent:FindFirstChild("ShooterDebug", true) end)
        if ok and result and result:IsA("BindableFunction") then return result end
    end
    return tryGet(LocalPlayer:FindFirstChild("PlayerScripts"))
        or tryGet(LocalPlayer:FindFirstChild("PlayerGui"))
        or tryGet(LocalPlayer.Character)
        or tryGet(LocalPlayer:FindFirstChild("Backpack"))
        or tryGet(LocalPlayer)
end

local function Invoke(...)
    if not ShooterDebug then return false, "ShooterDebug not found" end
    local ok, res = pcall(ShooterDebug.Invoke, ShooterDebug, ...)
    return ok, res
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude
rayParams.IgnoreWater = true

local function IsVisible(fromPos, toPos)
    local char = LocalPlayer.Character
    if not char then return false end
    rayParams.FilterDescendantsInstances = { char }
    local dir = toPos - fromPos
    if dir.Magnitude < 0.01 then return true end
    return Workspace:Raycast(fromPos, dir, rayParams) == nil
end

local function GetMyPos()
    local char = LocalPlayer.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    return hrp and hrp.Position or nil
end

local function FindBestTarget()
    local ok, state = Invoke("state")
    if not ok or not state or not state.ready then return nil end
    local npcCount = state.npcs or 0
    if npcCount == 0 then return nil end
    local myPos = GetMyPos()
    if not myPos then return nil end

    local bestIndex, bestDist = nil, Config.MaxDistance
    for i = 1, npcCount do
        local ok2, info = Invoke("npc", i)
        if ok2 and info and info.position then
            local aliveOk = (not Config.AliveCheck) or (info.alive == true)
            local consOk = (not Config.ConsciousCheck) or (info.conscious == true)
            if aliveOk and consOk then
                local dist = (info.position - myPos).Magnitude
                if dist < bestDist then
                    if (not Config.WallCheck) or IsVisible(myPos, info.position) then
                        bestDist = dist
                        bestIndex = i
                    end
                end
            end
        end
    end
    return bestIndex, bestDist
end

local lastFireAt = 0

local function FireOnce()
    local target, dist = FindBestTarget()
    if not target then return end
    if Config.FireMode == "Nade" and dist and dist < Config.NadeMinDist then return end

    if Config.FireMode == "Shoot" then
        Invoke("shoot", target, Config.TargetPart)
    elseif Config.FireMode == "Strike" then
        Invoke("strike", target, Config.TargetPart, Config.StrikeKind)
    elseif Config.FireMode == "Throw" then
        Invoke("throwat", target, Config.TargetPart)
    elseif Config.FireMode == "Nade" then
        Invoke("nade", target)
    end
end

local function ClearVelocities(model)
    for _, part in ipairs(model:GetDescendants()) do
        if part:IsA("BasePart") and not part.Anchored then
            part.AssemblyLinearVelocity = Vector3.zero
            part.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

local function PoolAll()
    if not ShooterDebug then return 0, "ShooterDebug not found" end
    local ok, state = Invoke("state")
    if not ok or not state or not state.ready then
        return 0, "Game not ready"
    end
    local npcCount = state.npcs or 0
    if npcCount == 0 then return 0, "No NPCs" end

    Invoke("pool", 1)
    task.wait(0.15)

    local ok0, firstInfo = Invoke("npc", 1)
    if not ok0 or not firstInfo or not firstInfo.position then
        return 0, "Calibration failed"
    end
    local basePos = firstInfo.position
    local moved = 1

    local radius = Config.PoolRadius
    for i = 2, npcCount do
        local ok2, info = Invoke("npc", i)
        if ok2 and info and info.alive and info.name then
            local model = Workspace:FindFirstChild(info.name)
            if model and model:IsA("Model") then
                local angle = ((i - 2) / math.max(npcCount - 1, 1)) * math.pi * 2
                local offset = Vector3.new(math.cos(angle) * radius, 0, math.sin(angle) * radius)
                local targetPos = basePos + offset
                local success = pcall(function()
                    model:PivotTo(CFrame.new(targetPos))
                    ClearVelocities(model)
                end)
                if success then
                    moved = moved + 1
                end
            end
        end
        task.wait(0.04)
    end

    return moved
end

task.spawn(function()
    while true do
        if Config.RagebotEnabled and ShooterDebug then
            local now = tick()
            if now - lastFireAt >= (Config.FireInterval / 1000) then
                pcall(FireOnce)
                lastFireAt = now
            end
        end
        task.wait(0.02)
    end
end)

    while true do
        task.wait(1)
        if ShooterDebug and Config.AutoReset then
            local ok, state = Invoke("state")
            if ok and state and state.ready then
                local aliveCount = 0
                for i = 1, (state.npcs or 0) do
                    local ok2, info = Invoke("npc", i)
                    if ok2 and info and info.alive then
                        aliveCount = aliveCount + 1
                    end
                end
                if aliveCount <= Config.ResetThreshold then
                    Invoke("reset")
                end
            end
        end
    end
end)

local autoPoolIndex = 0
task.spawn(function()
    while true do
        task.wait(0.8)
        if Config.AutoPool and ShooterDebug then
            local ok, state = Invoke("state")
            if ok and state and state.ready then
                local count = state.npcs or 0
                if count > 0 then
                    autoPoolIndex = autoPoolIndex + 1
                    if autoPoolIndex > count then autoPoolIndex = 1 end
                    pcall(Invoke, "pool", autoPoolIndex)
                end
            end
        end
    end
end)

local espHighlights = {}
local function SetESP(enabled)
    if enabled then
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") then
                if not Players:GetPlayerFromCharacter(obj) and not espHighlights[obj] then
                    local hl = Instance.new("Highlight")
                    hl.Name = "NPCEsp"
                    hl.Adornee = obj
                    hl.FillColor = Color3.fromRGB(255, 50, 50)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.FillTransparency = 0.5
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    hl.Parent = obj
                    espHighlights[obj] = hl
                end
            end
        end
    else
        for _, hl in pairs(espHighlights) do
            pcall(function() hl:Destroy() end)
        end
        espHighlights = {}
    end
end

task.spawn(function()
    while true do
        task.wait(2)
        if Config.ESP then SetESP(true) end
    end
end)

local function NukeAll()
    local ok, state = Invoke("state")
    if not ok or not state then return end
    local count = 0
    for i = 1, (state.npcs or 0) do
        local ok2, info = Invoke("npc", i)
        if ok2 and info and info.alive then
            Invoke("nade", i)
            count = count + 1
        end
    end
    Library:Notify("Dropped " .. count .. " grenades")
    task.delay(3, function() Invoke("boom") end)
end
