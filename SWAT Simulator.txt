local currentIndex = 1

-- ตัวแปรเก็บค่าการหมุนกล้องด้วยเมาส์
local cameraAngleX = 0
local cameraAngleY = 0

-- ดึงข้อมูล NPC ทั้งหมด
local function getNPCs()
	local npcs = {}
	for _, obj in ipairs(Workspace:GetChildren()) do
		if obj:IsA("Model") and not Players:GetPlayerFromCharacter(obj) then
			local humanoid = obj:FindFirstChildOfClass("Humanoid")
			if humanoid and humanoid.Health > 0 then
				table.insert(npcs, obj)
			end
		end
	end
	return npcs
end

local function stopSpectating()
	if renderConnection then
		renderConnection:Disconnect()
		renderConnection = nil
	end
	camera.CameraType = Enum.CameraType.Custom
end

local function spectateNextNPC()
	if not isSpectating then return end
	
	if renderConnection then
		renderConnection:Disconnect()
		renderConnection = nil
	end
	
	local npcList = getNPCs()
	
	if #npcList == 0 then
		stopSpectating()
		return
	end
	
	if currentIndex > #npcList then
		currentIndex = 1
	end
	
	local targetNPC = npcList[currentIndex]
	local humanoid = targetNPC and targetNPC:FindFirstChildOfClass("Humanoid")
	local head = targetNPC and (targetNPC:FindFirstChild("Head") or targetNPC:FindFirstChild("HumanoidRootPart"))
	
	if targetNPC and humanoid and humanoid.Health > 0 and head then
		camera.CameraType = Enum.CameraType.Scriptable
		
		renderConnection = RunService.RenderStepped:Connect(function(dt)
			if not isSpectating or not targetNPC or not head or not humanoid or humanoid.Health <= 0 then
				if renderConnection then renderConnection:Disconnect() end
				if isSpectating then
					currentIndex = currentIndex + 1
					spectateNextNPC()
				end
				return
			end
			
			-- คำนวณระยะห่างตาม Slider
			local distance = cameraDistance * 4 + 4
			
			-- สร้าง CFrame หมุนรอบตัว NPC ตามมุมกล้องที่ผู้เล่นเลื่อนเมาส์
			local rot = CFrame.Angles(0, cameraAngleX, 0) * CFrame.Angles(cameraAngleY, 0, 0)
			local offset = rot * Vector3.new(0, 3, distance)
			local targetPos = head.Position + offset
			
			camera.CFrame = CFrame.new(targetPos, head.Position)
		end)
		
		local diedConnection
		diedConnection = humanoid.Died:Connect(function()
			diedConnection:Disconnect()
			if renderConnection then renderConnection:Disconnect() end
			if isSpectating then
				currentIndex = currentIndex + 1
				spectateNextNPC()
			end
		end)
	else
		currentIndex = currentIndex + 1
		if isSpectating then
			spectateNextNPC()
		end
	end
end

-- ระบบรับค่าการขยับเมาส์เฉพาะตอนคลิกขวาค้างไว้ (เหมือนโหมดส่องทั่วไป) หรือจะปรับให้หมุนเลยก็ได้
UserInputService.InputChanged:Connect(function(input, gameProcessed)
	if isSpectating and input.UserInputType == Enum.UserInputType.MouseMovement then
		-- ตรวจสอบว่ากำลังกดคลิกขวาอยู่หรือไม่ (ถ้าต้องการให้หมุนได้เลยโดยไม่ต้องคลิกขวา ให้เอาเงื่อนไข MouseButton2 ออกได้)
		if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
			cameraAngleX = cameraAngleX - input.Delta.X * 0.005
			cameraAngleY = math.clamp(cameraAngleY - input.Delta.Y * 0.005, -math.pi/3, math.pi/3)
		end
	end
end)

local BlinkNPCSection = Tabs.Main:AddSection("Blink NPC")

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

local NPC_NAME = "NPC"
local SPACING = 3
local PLAYER_OFFSET = 5
local MOVE_INTERVAL = 0.25
local DRIFT_THRESHOLD = 0.8

local isFeatureEnabled = false
local originalCFrames = {}
local moveThread = nil
local lastCheckAt = 0

local function isAlive(npc)
    if not npc or not npc.Parent then
        return false
    end

    local humanoid = npc:FindFirstChildOfClass("Humanoid")
    return humanoid and humanoid.Health > 0
end

local function getActiveNPCs()
    local activeNPCs = {}
    local localCharacter = LocalPlayer.Character
    local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and obj.Name == NPC_NAME and isAlive(obj) then
            local npcRoot = obj:FindFirstChild("HumanoidRootPart")
            if npcRoot then
                table.insert(activeNPCs, obj)
            end
        end
    end

    table.sort(activeNPCs, function(a, b)
        local aRoot = a:FindFirstChild("HumanoidRootPart")
        local bRoot = b:FindFirstChild("HumanoidRootPart")
        if not localRoot then
            return false
        end
        if not aRoot or not bRoot then
            return aRoot ~= nil
        end
        return (aRoot.Position - localRoot.Position).Magnitude < (bRoot.Position - localRoot.Position).Magnitude
    end)

    return activeNPCs
end

local function restoreOriginalNPCs()
    for npc, originalCFrame in pairs(originalCFrames) do
        if npc and npc.Parent and npc:FindFirstChild("HumanoidRootPart") and isAlive(npc) then
            npc.HumanoidRootPart.CFrame = originalCFrame
        end
    end
    originalCFrames = {}
end

local function updateNPCRow(force)
    if not isFeatureEnabled then
        return
    end

    local character = LocalPlayer.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then
        return
    end

    local hrp = character.HumanoidRootPart
    local now = os.clock()
    if not force and now - lastCheckAt < MOVE_INTERVAL then
        return
    end
    lastCheckAt = now

    local activeNPCs = getActiveNPCs()
    if #activeNPCs == 0 then
        return
    end

    local forwardVector = hrp.CFrame.LookVector
    local rowCenter = hrp.Position

    for i, npc in ipairs(activeNPCs) do
        local npcRoot = npc:FindFirstChild("HumanoidRootPart")
        if npcRoot then
            if not originalCFrames[npc] then
                originalCFrames[npc] = npcRoot.CFrame
            end

            local offsetDistance = PLAYER_OFFSET + ((i - 1) * SPACING)
            local targetPos = rowCenter + (forwardVector * offsetDistance)
            targetPos = Vector3.new(targetPos.X, rowCenter.Y + 1, targetPos.Z)

            local desiredCFrame = CFrame.new(targetPos, targetPos + forwardVector)
            local distanceFromTarget = (npcRoot.Position - targetPos).Magnitude

            if force or distanceFromTarget > DRIFT_THRESHOLD then
                npcRoot.CFrame = desiredCFrame
            end

            if npcRoot.AssemblyLinearVelocity.Magnitude > 0.5 then
                npcRoot.AssemblyLinearVelocity = Vector3.zero
                npcRoot.AssemblyAngularVelocity = Vector3.zero
            end
        end
    end
end

local function stopFeature()
    if moveThread then
        pcall(function()
            task.cancel(moveThread)
        end)
        moveThread = nil
    end
    lastCheckAt = 0
    restoreOriginalNPCs()
end

local Toggle = Tabs.Main:AddToggle("BringNPCToggle", {
    Title = "ดึง NPC เรียงหน้าจอ",
    Default = false
})

Toggle:OnChanged(function(Value)
    isFeatureEnabled = Value

    if isFeatureEnabled then
        if moveThread then
            pcall(function()
                task.cancel(moveThread)
            end)
            moveThread = nil
        end

        updateNPCRow(true)

        moveThread = task.spawn(function()
            while isFeatureEnabled do
                updateNPCRow(false)
                task.wait(MOVE_INTERVAL)
            end
        end)
    else
        stopFeature()
    end
end)

-- ดักจับ Token และข้อมูลปืน
local oldNamecall
local function safeHook()
    if not hookmetamethod then
        return false
    end
    oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        local args = {...}

        if self == damageEvent and (method == "InvokeServer" or method == "FireServer") then
            local tokenArg = args[#args]
            if type(tokenArg) == "string" and string.find(tokenArg, "-") then
                getgenv().RealToken = tokenArg
            end

            if type(args[5]) == "table" then
                getgenv().LastGunArgs = args[5]
            end
            if type(args[6]) == "table" then
                getgenv().LastAttachments = args[6]
            end
        end

        if oldNamecall then
            return oldNamecall(self, ...)
        end
        return nil
    end)
    return true
end

if damageEvent and not safeHook() then
    warn("[AuraKill] hookmetamethod not supported on this executor.")
end

Toggle:OnChanged(function()
    getgenv().AuraKillAll = Toggle.Value
    
    if getgenv().AuraKillAll then
        if not getgenv().RealToken or not getgenv().LastGunArgs then

local OpenDoorSection = Tabs.Main:AddSection("Open Door")

local originalHoldDurations = {}
local isRunning = false

local Toggle = Tabs.Main:AddToggle("InstantPromptToggle", {Title = "Instant Open Door Fast", Default = false })

Toggle:OnChanged(function()
    isRunning = Toggle.Value
    
    if isRunning then
        task.spawn(function()
            local cachedDoorsFolder = nil
            
            -- ค้นหาตำแหน่งโฟลเดอร์ Doors แค่ครั้งเดียวตอนเปิด Toggle เพื่อความลื่นไหล
            for _, obj in ipairs(workspace:GetDescendants()) do
                if obj:IsA("Folder") and obj.Name == "Doors" then
                    local hasEnterableDoor = false
                    for _, child in ipairs(obj:GetChildren()) do
                        if child.Name == "EnterableDoor" then
                            hasEnterableDoor = true
                            break
                        end
                    end
                    if hasEnterableDoor then
                        cachedDoorsFolder = obj
                        break
                    end
                end
            end
            
            -- วนลูปทำงานเฉพาะกับโฟลเดอร์ที่เจอแล้ว (ไม่เปลืองแรงค้นหาใหม่ทุกรอบ)
            while isRunning do
                if cachedDoorsFolder and cachedDoorsFolder.Parent then
                    for _, door in ipairs(cachedDoorsFolder:GetChildren()) do
                        if door.Name == "EnterableDoor" then
                            local hingePart = door:FindFirstChild("HingePart", true)
                            if hingePart then
                                local prompts = hingePart:FindFirstChild("Prompts", true) or hingePart
                                for _, promptChild in ipairs(prompts:GetDescendants()) do
                                    if promptChild:IsA("ProximityPrompt") then
                                        if originalHoldDurations[promptChild] == nil then
                                            originalHoldDurations[promptChild] = promptChild.HoldDuration
                                        end
                                        if promptChild.HoldDuration ~= 0 then
                                            promptChild.HoldDuration = 0
                                        end
                                    end
                                end
                            end
                        end
                    end
                else
                    -- เผื่อกรณีหาไม่เจอรอบแรก ลองหาใหม่อีกครั้ง
                    for _, obj in ipairs(workspace:GetDescendants()) do
                        if obj:IsA("Folder") and obj.Name == "Doors" then
                            for _, child in ipairs(obj:GetChildren()) do
                                if child.Name == "EnterableDoor" then
                                    cachedDoorsFolder = obj
                                    break
                                end
                            end
                            if cachedDoorsFolder then break end
                        end
                    end
                end
                
                -- หน่วงเวลา 1 วินาทีแบบสบายๆ ไม่กินเครื่อง
                task.wait(1)
            end
        end)
    else
        -- คืนค่า HoldDuration เดิมกลับทั้งหมดเมื่อปิด Toggle
        for promptChild, originalDuration in pairs(originalHoldDurations) do
            if promptChild and promptChild.Parent then
                promptChild.HoldDuration = originalDuration
            end
        end
        originalHoldDurations = {}
    end
end)

    if ESPDeathHooks[model] then
        pcall(function()
            ESPDeathHooks[model]:Disconnect()
        end)
        ESPDeathHooks[model] = nil
    end

    if ESPCache[model] then
        for _, drawing in pairs(ESPCache[model]) do
            if type(drawing) == "table" then
                for _, line in pairs(drawing) do
                    if line and line.Visible ~= nil then
                        pcall(function() line.Visible = false end)
                    end
                    if line and line.Remove then
                        pcall(function() line:Remove() end)
                    end
                end
            elseif drawing and drawing.Visible ~= nil then
                pcall(function() drawing.Visible = false end)
            end

            if drawing and drawing.Remove then
                pcall(function() drawing:Remove() end)
            end
        end
        ESPCache[model] = nil
    end
end

local function ClearAllESP()
    for model in pairs(ESPCache) do
        RemoveESP(model)
    end
    ESPCache = {}
end

local function HookNPCDeath(model)
    if not model or ESPDeathHooks[model] then
        return
    end

    local humanoid = model:FindFirstChildOfClass("Humanoid")
    if not humanoid then
        return
    end

    ESPDeathHooks[model] = humanoid.Died:Connect(function()
        if ESPCache[model] then
            RemoveESP(model)
        end
    end)
end

local function CreateDrawing(type, props)
    local obj = Drawing.new(type)
    for i, v in pairs(props) do obj[i] = v end
    return obj
end

local Bones = {
    {"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"},
    {"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"},
    {"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
    {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
    {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"}
}

local function DrawSkeleton(cache, model, currentCamera)
    if not cache.SkeletonLines then cache.SkeletonLines = {} end

    local headPart = model:FindFirstChild("Head")
    if headPart then
        if not cache.HeadCircle then
            cache.HeadCircle = CreateDrawing("Circle", { Filled = false, Thickness = 1.5, Color = Color3.fromRGB(128, 255, 0), Transparency = 0.8 })
        end
        local headPos, headOnScreen = currentCamera:WorldToViewportPoint(headPart.Position)
        local topHeadPos = currentCamera:WorldToViewportPoint(headPart.Position + Vector3.new(0, 0.8, 0))
        local headRadius = math.abs(headPos.Y - topHeadPos.Y) * 0.75

        if headOnScreen then
            cache.HeadCircle.Position = Vector2.new(headPos.X, headPos.Y)
            cache.HeadCircle.Radius = math.clamp(headRadius, 3, 12)
            cache.HeadCircle.Visible = true
        else
            cache.HeadCircle.Visible = false
        end
    else
        if cache.HeadCircle then cache.HeadCircle.Visible = false end
    end

    for i, pair in ipairs(Bones) do
        local partA = model:FindFirstChild(pair[1])
        local partB = model:FindFirstChild(pair[2])
        local line = cache.SkeletonLines[i]

        if not line then
            line = CreateDrawing("Line", { Thickness = 1.5, Color = Color3.fromRGB(128, 255, 0), Transparency = 0.8 })
            cache.SkeletonLines[i] = line
        end

        if partA and partB then
            local posA, onScreenA = currentCamera:WorldToViewportPoint(partA.Position)
            local posB, onScreenB = currentCamera:WorldToViewportPoint(partB.Position)
            if onScreenA or onScreenB then
                line.From = Vector2.new(posA.X, posA.Y)
                line.To = Vector2.new(posB.X, posB.Y)
                line.Visible = true
            else
                line.Visible = false
            end
        else
            line.Visible = false
        end
    end
end

local function HideSkeleton(cache)
    if cache.HeadCircle then cache.HeadCircle.Visible = false end
    if cache.SkeletonLines then
        for _, line in pairs(cache.SkeletonLines) do
            if line then line.Visible = false end
        end
    end
end

local function UpdateESP()
    if not ESPRunning then
        ClearAllESP()
        return
    end

    local currentCamera = Workspace.CurrentCamera
    if not currentCamera then
        return
    end

    local activeModels = {}
    local localCharacter = LocalPlayer and LocalPlayer.Character
    local localRoot = localCharacter and (localCharacter:FindFirstChild("HumanoidRootPart") or localCharacter:FindFirstChild("Head"))

    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name == "NPC" then
            local humanoid = model:FindFirstChildOfClass("Humanoid")
            local rootPart = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Head") or model:FindFirstChild("LowerTorso")

            if humanoid and humanoid.Health <= 0 then
                if ESPCache[model] then
                    RemoveESP(model)
                end
            elseif humanoid and humanoid.Health > 0 and rootPart then
                activeModels[model] = true

                if not ESPCache[model] then
                    HookNPCDeath(model)
                    ESPCache[model] = {
                        Name = CreateDrawing("Text", { Size = 14, Center = true, Outline = true, Color = Color3.new(1,1,1) }),
                        Hp = CreateDrawing("Text", { Size = 14, Center = true, Outline = true }),
                        Line = CreateDrawing("Line", { Thickness = 1, Color = Color3.fromRGB(255, 255, 0), Transparency = 0.7 }),
                        Box = CreateDrawing("Square", { Filled = false, Thickness = 1.5, Color = Color3.new(0.2, 0.6, 1) }),
                        Weapon = CreateDrawing("Text", { Size = 13, Center = true, Outline = true, Color = Color3.new(0.9, 0.9, 0.9) })
                    }
                end

                local cache = ESPCache[model]
                local vector, onScreen = currentCamera:WorldToViewportPoint(rootPart.Position)

                if onScreen then
                    local head = model:FindFirstChild("Head") or rootPart
                    local topVector = currentCamera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                    local bottomVector = currentCamera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, 2.5, 0))
                    local height = math.abs(topVector.Y - bottomVector.Y)
                    local width = height * 0.6
                    local boxPos = Vector2.new(vector.X - width / 2, topVector.Y)
                    local boxSize = Vector2.new(width, height)

                    if ESPSettings["Bounding Box"] then
                        cache.Box.Visible = true
                        cache.Box.Size = boxSize
                        cache.Box.Position = boxPos
                    else
                        cache.Box.Visible = false
                    end

                    if ESPSettings["Skeleton"] then
                        DrawSkeleton(cache, model, currentCamera)
                    else
                        HideSkeleton(cache)
                    end

                    local distText = ""
                    if ESPSettings["Distance"] and localCharacter and localRoot then
                        local distance = (rootPart.Position - localRoot.Position).Magnitude
                        distText = " [" .. math.floor(distance) .. "m]"
                    end

                    local offset = 0
                    local bottomCenter = Vector2.new(vector.X, bottomVector.Y + 5)

                    if ESPSettings["Name Tag"] then
                        cache.Name.Text = model.Name .. distText
                        cache.Name.Position = Vector2.new(bottomCenter.X, bottomCenter.Y + offset)
                        cache.Name.Visible = true
                        offset = offset + 15
                    elseif ESPSettings["Distance"] then
                        cache.Name.Text = distText:gsub("^%s*", "")
                        cache.Name.Position = Vector2.new(bottomCenter.X, bottomCenter.Y + offset)
                        cache.Name.Visible = true
                        offset = offset + 15
                    else
                        cache.Name.Visible = false
                    end

                    if ESPSettings["Health Bar"] then
                        local hpPercent = humanoid.Health / humanoid.MaxHealth
                        local hpColor = hpPercent > 0.5 and Color3.fromRGB(0, 255, 0) or (hpPercent > 0.25 and Color3.fromRGB(255, 255, 0) or Color3.fromRGB(255, 0, 0))
                        cache.Hp.Text = "HP: " .. math.floor(humanoid.Health) .. " / " .. math.floor(humanoid.MaxHealth)
                        cache.Hp.Color = hpColor
                        cache.Hp.Position = Vector2.new(bottomCenter.X, bottomCenter.Y + offset)
                        cache.Hp.Visible = true
                        offset = offset + 15
                    else
                        cache.Hp.Visible = false
                    end

                    if ESPSettings["Weapon Info"] then
                        local weaponName = "Unarmed"
                        for _, item in ipairs(model:GetChildren()) do
                            if item:IsA("Tool") then
                                weaponName = item.Name
                                break
                            end
                        end
                        cache.Weapon.Text = "[" .. weaponName .. "]"
                        cache.Weapon.Position = Vector2.new(bottomCenter.X, bottomCenter.Y + offset)
                        cache.Weapon.Visible = true
                    else
                        cache.Weapon.Visible = false
                    end

                    if ESPSettings["Snap Lines"] then
                        cache.Line.Visible = true
                        local startPos = Vector2.new(currentCamera.ViewportSize.X / 2, currentCamera.ViewportSize.Y)
                        if SnapLinePosition == "Top" then
                            startPos = Vector2.new(currentCamera.ViewportSize.X / 2, 0)
                        elseif SnapLinePosition == "Mid" then
                            startPos = Vector2.new(currentCamera.ViewportSize.X / 2, currentCamera.ViewportSize.Y / 2)
                        elseif SnapLinePosition == "Down" then
                            startPos = Vector2.new(currentCamera.ViewportSize.X / 2, currentCamera.ViewportSize.Y)
                        end

                        cache.Line.From = startPos
                        cache.Line.To = Vector2.new(vector.X, bottomVector.Y)
                    else
                        cache.Line.Visible = false
                    end
                else
                    cache.Box.Visible = false
                    cache.Name.Visible = false
                    cache.Hp.Visible = false
                    cache.Line.Visible = false
                    cache.Weapon.Visible = false
                    HideSkeleton(cache)
                end
            else
                if ESPCache[model] then
                    RemoveESP(model)
                end
            end
        end
    end

    for model in pairs(ESPCache) do
        if not activeModels[model] then
            RemoveESP(model)
        end
    end
end

local AimbotSettings = {
    Enabled = false,
    TargetPart = "Head",
    ShowFOV = false,
    FOVRadius = 150,
    WallCheck = false -- ค่าเริ่มต้นปิดไว้

local FOVCircle = Drawing.new("Circle")
FOVCircle.Visible = false
FOVCircle.Color = Color3.fromRGB(255, 255, 255)
FOVCircle.Thickness = 1
FOVCircle.Filled = false
FOVCircle.Transparency = 1

-- ฟังก์ชันสำหรับเช็คว่ามองเห็นเป้าหมายหรือไม่ (แก้ปัญหา Raycast และเพิ่มความเสถียร)
local function IsVisible(targetPart, targetModel)
    if not AimbotSettings.WallCheck then return true end
    if not targetPart or not targetModel then return false end
    
    local cameraPos = Camera.CFrame.Position
    local direction = (targetPart.Position - cameraPos)
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    
    -- รวบรวมรายชื่อสิ่งที่ต้องการละเว้นไม่ให้ Raycast ชน
    local ignoreList = {}
    if LocalPlayer.Character then
        table.insert(ignoreList, LocalPlayer.Character)
    end
    
    if targetModel then
        table.insert(ignoreList, targetModel)
    end
    
    raycastParams.FilterDescendantsInstances = ignoreList
    raycastParams.IgnoreWater = true
    
    local raycastResult = Workspace:Raycast(cameraPos, direction, raycastParams)
    
    -- ถ้า Raycast ชนวัตถุอื่นก่อนถึงเป้าหมาย แสดงว่ามีกำแพงบัง
    if raycastResult then
        local hitInstance = raycastResult.Instance
        if hitInstance and not hitInstance:IsDescendantOf(targetModel) then
            return false
        end
    end
    
    return true
end

-- Loop การทำงานของ Aimbot (ปรับปรุงแก้ปัญหาหลุดล็อกเวลาขยับเมาส์แรงๆ)
RunService.RenderStepped:Connect(function()
    local mouseLoc = UserInputService:GetMouseLocation()
    
    -- อัปเดตตำแหน่งและขนาดของ FOV ตามเมาส์
    if AimbotSettings.ShowFOV then
        FOVCircle.Position = mouseLoc
        FOVCircle.Radius = tonumber(AimbotSettings.FOVRadius) or 150
        FOVCircle.Visible = true
    else
        FOVCircle.Visible = false
    end

    -- เช็คการกดคลิกขวาค้างเพื่อล็อกเป้า
    if AimbotSettings.Enabled and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        
        -- ตรวจสอบว่าเป้าหมายปัจจุบันยังอยู่ดี, HP > 0
        if currentTarget then
            local model = currentTarget.Parent
            local humanoid = model and model:FindFirstChildOfClass("Humanoid")
            local health = humanoid and tonumber(humanoid.Health) or 0
            
            -- ถ้าตัวละครตาย หรือโมเดลหายไป ให้ยกเลิกการล็อก
            if not humanoid or health <= 0 or not model:IsDescendantOf(Workspace) then
                currentTarget = nil
            elseif AimbotSettings.WallCheck then
                -- เช็ค WallCheck สำหรับเป้าหมายที่ล็อกอยู่ (ถ้าถูกบังมิดชิดค่อยหลุด)
                if not IsVisible(currentTarget, model) then
                    currentTarget = nil
                end
            end
        end

        -- ถ้ายังไม่มีเป้าหมาย ให้หาเป้าหมายใหม่ที่อยู่ใน FOV และมองเห็นได้
        if not currentTarget then
            local shortestDistance = tonumber(AimbotSettings.FOVRadius) or 150
            local bestTarget = nil

            for _, model in ipairs(Workspace:GetChildren()) do
                if model:IsA("Model") and model.Name == "NPC" then
                    local humanoid = model:FindFirstChildOfClass("Humanoid")
                    local rootPart = model:FindFirstChild("HumanoidRootPart")

                    if humanoid and rootPart then
                        local health = tonumber(humanoid.Health) or 0
                        if health > 0 then
                            local partToLock = model:FindFirstChild("Head")
                            if AimbotSettings.TargetPart == "Body" then
                                partToLock = model:FindFirstChild("UpperTorso") or model:FindFirstChild("Torso") or rootPart
                            end

                            if partToLock and IsVisible(partToLock, model) then
                                local screenPoint, onScreen = Camera:WorldToViewportPoint(partToLock.Position)
                                if onScreen then
                                    local screenPos = Vector2.new(screenPoint.X, screenPoint.Y)
                                    local distance = (screenPos - mouseLoc).Magnitude

                                    if distance < shortestDistance then
                                        shortestDistance = distance
                                        bestTarget = partToLock
                                    end
                                end
                            end
                        end
                    end
                end
            end
            
            currentTarget = bestTarget
        end

        -- ถ้ามีเป้าหมาย ให้ล็อกกล้อง (ต่อให้สะบัดเมาส์แรงแค่ไหน เป้าหมายก็จะไม่หลุดออกจาก FOV ระหว่างล็อก)
        if currentTarget then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, currentTarget.Position)
        end
    else
        currentTarget = nil
    end
end)

        end)

        end

    _G.__EVIL_QT = MOD

    function MOD.cleanup()
        for _, c in ipairs(MOD.conns) do
            pcall(function() c:Disconnect() end)
        end
        if MOD.gui and MOD.gui.Parent then
            pcall(function() MOD.gui:Destroy() end)
        end
        _G.__EVIL_QT = nil
    end

        local ok, host = pcall(function()

        end)
        if ok and host then return host end
        local pg = Players.LocalPlayer:FindFirstChildOfClass("PlayerGui") or Players.LocalPlayer:WaitForChild("PlayerGui")
        return pg
    end

    quickGui.Name = "FluentQuickToggle"

    quickGui.DisplayOrder = 9999
    pcall(function() if syn and syn.protect_gui then syn.protect_gui(quickGui) end end)
    quickGui.Parent = getGuiParent()
    MOD.gui = quickGui

    local evilButton = Instance.new("TextButton")
    evilButton.Name = "Evil"
    evilButton.Parent = quickGui
    evilButton.Size = UDim2.fromOffset(120, 40)
    evilButton.Position = UDim2.fromOffset(20, 120)

        pcall(function()

                Title = title or "Info", Text = text or "", Duration = dur or 2
            })
        end)
    end

        if typeof(Window) == "table" and Window.Minimize then
            Window:Minimize()
            notify("Fluent", "Toggled", 2)
        else
            notify("Fluent", "Window not ready", 2)
        end
    end

    evilButton.AnchorPoint = Vector2.new(0, 0)
    local dragging, dragStartPos, startOffset = false, nil, nil
    local movedPixels = 0
    local CLICK_MOVE_THRESHOLD = 6

    local function clampToViewport(px, py)
        local vs = (Camera and Camera.ViewportSize) or Vector2.new(1280, 720)
        local bw, bh = evilButton.AbsoluteSize.X, evilButton.AbsoluteSize.Y
        local minX, minY = 0, 0
        local maxX, maxY = vs.X - bw, vs.Y - bh
        return math.clamp(px, minX, maxX), math.clamp(py, minY, maxY)
    end

    local function beginDrag(startInput)
        dragging = true
        movedPixels = 0
        dragStartPos = UIS:GetMouseLocation()
        startOffset  = evilButton.Position

        table.insert(MOD.conns, UIS.InputChanged:Connect(function(input)
            if not dragging then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
                local now = UIS:GetMouseLocation()
                local delta = now - dragStartPos
                movedPixels = math.max(movedPixels, delta.Magnitude)

                local newX = startOffset.X.Offset + delta.X
                local newY = startOffset.Y.Offset + delta.Y
                local cx, cy = clampToViewport(newX, newY)
                evilButton.Position = UDim2.fromOffset(cx, cy)
            end
        end))

        local endConn
        endConn = startInput.Changed:Connect(function()
            if startInput.UserInputState == Enum.UserInputState.End then
                dragging = false
                pcall(function() endConn:Disconnect() end)
                if movedPixels < CLICK_MOVE_THRESHOLD then
                    toggleFluent()
                end
            end
        end)
        table.insert(MOD.conns, endConn)
    end

    table.insert(MOD.conns, evilButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            beginDrag(input)
        end
    end))

    print("[Evil] ready. GUI:", quickGui.Parent, "Pos:", evilButton.Position)
end

local function processtext(text)
    if string.find(text, oldName) then
        return string.gsub(text, oldName, getgenv().Config.FakeName)
    elseif string.find(text, oldDisplayName) then
        return string.gsub(text, oldDisplayName, getgenv().Config.FakeDisplayName)
    end
    return text
end
