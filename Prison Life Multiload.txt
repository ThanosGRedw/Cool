-- Load WindUI Library using the official release URL
local _version = "1.6.66"
local WindUI = loadstring(game:HttpGet("https://github.com/Footagesus/WindUI/releases/download/" .. _version .. "/main.lua"))()

-- Create Main Window
local Window = WindUI:CreateWindow({
    Title = "Prison life ezload",
    Icon = "sparkles",
    Author = "goathanos",
    Folder = "WindUI_Config",
    Size = UDim2.fromOffset(580, 460),
    Transparent = true,
    Theme = "Dark",
    SideBarWidth = 170,
    HasOutline = true,
})

-- Create Tab
local MainTab = Window:Tab({
    Title = "Main",
    Icon = "home",
})

-- BUTTON 1 (Added directly to the tab)
MainTab:Button({
    Title = "Precision hub",
    Desc = "best script",
    Callback = function()
loadstring(game:HttpGet("https://raw.githubusercontent.com/ThanosGRedw/Cool/refs/heads/main/Precision%20Hub.txt"))()
        WindUI:Notify({
            Title = "Executed",
            Content = "Executed successfully.",
            Duration = 3,
            Icon = "check"
        })
    end,
})

-- BUTTON 2
MainTab:Button({
    Title = "Kx admin",
    Desc = "Kx admin",
    Callback = function()
loadstring(game:HttpGet("https://raw.githubusercontent.com/Keraunius/213sadd23/refs/heads/main/kxadm.lua"))()
        WindUI:Notify({
            Title = "Executed",
            Content = "Executed successfully.",
            Duration = 3,
            Icon = "check"
        })
    end,
})

-- BUTTON 3
MainTab:Button({
    Title = "empty",
    Desc = "empty.",
    Callback = function()
        print("hi"
        WindUI:Notify({
            Title = "Executed",
            Content = "Executed successfully.",
            Duration = 3,
            Icon = "check"
        })

})
