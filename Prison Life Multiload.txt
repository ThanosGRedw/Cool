-- Load WindUI Library using the official release URL
local _version = "1.6.66"
local WindUI = loadstring(game:HttpGet("https://github.com/Footagesus/WindUI/releases/download/" .. _version .. "/main.lua"))()

-- Create Main Window
local Window = WindUI:CreateWindow({
    Title = "Prison Life ezload",
    Icon = "sparkles",
    Author = "by goathanos",
    Folder = "WindUI_Config",
    Size = UDim2.fromOffset(580, 460),
    Transparent = false,
    Theme = "Midnight",
    SideBarWidth = 170,
    HasOutline = true,
})

-- Create Tabs
local MainTab = Window:Tab({
    Title = "Main",
    Icon = "house",
})

local SettingsTab = Window:Tab({
    Title = "Settings",
    Icon = "settings",
})

-- ==================== MAIN TAB BUTTONS ====================

MainTab:Button({
    Title = "Precision Hub",
    Callback = function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/ThanosGRedw/Cool/refs/heads/main/Precision%20Hub.txt"))()
        WindUI:Notify({
            Title = "Executed",
            Content = "Precision Hub executed successfully.",
            Duration = 3,
            Icon = "check"
        })
    end,
})

-- ==================== SETTINGS TAB ====================

-- Native WindUI Transparency Slider (Moves smoothly by 1)
SettingsTab:Slider({
    Title = "Transparency Level",
    Desc = "0 = Solid, 100 = Max Transparency",
    Step = 1,
    Value = {
        Min = 0,
        Max = 100,
        Default = 0,
    },
    Callback = function(value)
        local alpha = value / 100
        WindUI.TransparencyValue = alpha
        
        if value > 0 then
            Window:ToggleTransparency(true)
        else
            Window:ToggleTransparency(false)
        end
    end,
})

-- Select initial tab
Window:SelectTab(1)
