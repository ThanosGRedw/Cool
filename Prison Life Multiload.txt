-- Load WindUI Library using the official release URL
local _version = "1.6.66"
local WindUI = loadstring(game:HttpGet("https://github.com/Footagesus/WindUI/releases/download/" .. _version .. "/main.lua"))()

-- Create Main Window
local Window = WindUI:CreateWindow({
    Title = "Prison Life ezload",
    Icon = "sparkles",
    Author = "by goathanos",
    Folder = "WindUI_Config",
    Size = UDim2.fromOffset(580, 460),
    Transparent = false,
    Theme = "Midnight",
    SideBarWidth = 170,
    HasOutline = true,
})

-- Create Tabs
local MainTab = Window:Tab({
    Title = "Main",
    Icon = "home",
})

local SettingsTab = Window:Tab({
    Title = "Settings",
    Icon = "settings",
})

-- ==================== MAIN TAB BUTTONS ====================

-- BUTTON 1
MainTab:Button({
    Title = "Precision Hub",
    Callback = function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/ThanosGRedw/Cool/refs/heads/main/Precision%20Hub.txt"))()
        WindUI:Notify({
            Title = "Executed",
            Content = "Precision Hub executed successfully.",
            Duration = 3,
            Icon = "check"
        })
    end,
})

-- BUTTON 2
MainTab:Button({
    Title = "KX Admin",
    Callback = function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/Keraunius/213sadd23/refs/heads/main/kxadm.lua"))()
        WindUI:Notify({
            Title = "Executed",
            Content = "KX Admin executed successfully.",
            Duration = 3,
            Icon = "check"
        })
    end,
})

-- BUTTON 3
MainTab:Button({
    Title = "Parallax",
    Callback = function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/ThanosGRedw/Cool/refs/heads/main/parallax.txt"))()
        WindUI:Notify({
            Title = "Executed",
            Content = "Parallax executed successfully.",
            Duration = 3,
            Icon = "check"

    end,

-- BUTTON 4
MainTab:Button({
    Title = "Silly Scripts",

            Content = "Silly Scripts executed successfully.",
            Duration = 3,

        })
    end,
})

-- ==================== SETTINGS TAB ====================

-- Toggle Transparency
SettingsTab:Toggle({
    Title = "Window Transparency",
    Desc = "Toggle background transparency on or off.",
    Default = false,
    Callback = function(state)
        Window:ToggleTransparency(state)
        WindUI:Notify({
            Title = "Settings Updated",
            Content = "Transparency set to: " .. tostring(state),
            Duration = 2,
            Icon = "sliders"
        })
    end,
})

-- Theme Selector Dropdown
SettingsTab:Dropdown({
    Title = "UI Theme",
    Desc = "Change the look of the script hub.",
    Values = {"Midnight", "Dark", "Light", "Darker"},
    Default = "Midnight",
    Callback = function(selectedTheme)
        Window:SetTheme(selectedTheme)
        WindUI:Notify({
            Title = "Theme Changed",
            Content = "Switched theme to " .. selectedTheme,
            Duration = 2,
            Icon = "palette"
        })
    end,
})

-- Select initial tab
Window:SelectTab(1)
