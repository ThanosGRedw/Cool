local function getMonster()
    local entities = workspace:FindFirstChild("Entities")
    if not entities then return nil end
    local locust = entities:FindFirstChild("Locust")
    if locust and locust:IsA("Model") and locust:FindFirstChildOfClass("Humanoid") then
        return locust
    end
    for _, v in ipairs(entities:GetChildren()) do
        if v:IsA("Model") and v:FindFirstChildOfClass("Humanoid") then return v end
    end
    return nil
end

local function getAnchor(inst)
    if not inst then return nil end
    if inst:IsA("BasePart") then return inst end
    if inst:IsA("Tool") then return inst:FindFirstChild("Handle") or inst:FindFirstChildWhichIsA("BasePart") end
    if inst:IsA("Model") then
        return inst.PrimaryPart
            or inst:FindFirstChild("Interact")
            or inst:FindFirstChild("HumanoidRootPart")
            or inst:FindFirstChild("Head")
            or inst:FindFirstChildWhichIsA("BasePart")
    end
    return nil
end

local function cabinetPrompt(model)
    local inter = model:FindFirstChild("Interact")
    return inter and inter:FindFirstChildOfClass("ProximityPrompt")
end

-- The game NEVER disables a searched cabinet's prompt (verified: prompt stays
-- Enabled after the server accepts a search), so "searched" cannot be read off
-- the game. It is only what the auto-escape has already been through.
_G.__LustSearched = _G.__LustSearched or {}
local function cabinetSearched(model)
    local pp = cabinetPrompt(model)
    if pp == nil then return true end
    return _G.__LustSearched[model] == true
end

local Tags = {}      -- [anchorPart] = { hl, bb, lbl, used }
local Tracer = nil

local function makeTag()
    local hl = Instance.new("Highlight")
    hl.FillTransparency = 0.65
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Enabled = false
    hl.Parent = ScreenParent

    local bb = Instance.new("BillboardGui")
    bb.Size = UDim2.fromOffset(240, 32)
    bb.StudsOffsetWorldSpace = Vector3.new(0, 3, 0)
    bb.AlwaysOnTop = true
    bb.MaxDistance = math.huge
    bb.Enabled = false
    bb.Parent = ScreenParent

    local lbl = Instance.new("TextLabel")
    lbl.BackgroundTransparency = 1
    lbl.Size = UDim2.fromScale(1, 1)
    lbl.Font = Enum.Font.GothamBold
    lbl.TextSize = 14
    lbl.TextStrokeTransparency = 0.35
    lbl.TextStrokeColor3 = Color3.new(0, 0, 0)
    lbl.Parent = bb

    return { hl = hl, bb = bb, lbl = lbl }
end

-- adornee: what the Highlight wraps (model or part). anchor: what the label sits on.
local function drawTag(anchor, adornee, text, color, chams)
    local tag = Tags[anchor]
    if not tag then
        tag = makeTag()
        Tags[anchor] = tag
    end
    tag.used = true

    tag.hl.Adornee = adornee
    tag.hl.FillColor = color
    tag.hl.OutlineColor = color
    tag.hl.Enabled = chams

    tag.bb.Adornee = anchor
    tag.bb.Enabled = true
    tag.lbl.Text = text
    tag.lbl.TextColor3 = color
end

local function sweepUnused()
    for anchor, tag in pairs(Tags) do
        if not tag.used then
            tag.hl:Destroy()
            tag.bb:Destroy()
            Tags[anchor] = nil
        else
            tag.used = false
        end
    end
end

local function clearAll()
    for anchor, tag in pairs(Tags) do
        tag.hl:Destroy()
        tag.bb:Destroy()
        Tags[anchor] = nil
    end
    if Tracer then Tracer.Visible = false end
end

local function label(name, dist)
    if S.ShowDistance then
        return string.format("%s  [%dm]", name, math.floor(dist + 0.5))
    end
    return name
end

local function inRange(dist)
    return S.MaxDistance <= 0 or dist <= S.MaxDistance
end

--=========================================================================--
--                              REFRESH (15 Hz)                            --
--=========================================================================--

local function refresh()
    local eye = Camera.CFrame.Position

    ------------------------------------------------ monster
    if S.Monster then
        local m = getMonster()
        local a = getAnchor(m)
        if a then
            local d = (a.Position - eye).Magnitude
            if inRange(d) then
                local head = m:FindFirstChild("Head") or a
                drawTag(head, m, label("LOCUST", d), S.MonsterColor, S.MonsterChams)
            end
        end
    end

    ------------------------------------------------ cabinets
    if S.Cabinets then
        local folder = workspace:FindFirstChild("Cabinets")
        if folder then
            local list = {}
            for _, c in ipairs(folder:GetChildren()) do
                local a = getAnchor(c)
                if a then
                    local d = (a.Position - eye).Magnitude
                    local searched = cabinetSearched(c)
                    if inRange(d) and not (searched and S.HideSearched) then
                        table.insert(list, { model = c, anchor = a, dist = d, searched = searched })
                    end
                end
            end
            table.sort(list, function(x, y) return x.dist < y.dist end)

            for i, e in ipairs(list) do
                local color = e.searched and DIM or S.CabinetColor
                local name = e.searched and "CABINET (searched)" or "CABINET"
                local chams = (not e.searched) and (i <= S.CabinetChams)
                drawTag(e.anchor, e.model, label(name, e.dist), color, chams)
            end
        end
    end

    ------------------------------------------------ objectives
    if S.Objectives then
        for _, o in ipairs(OBJECTIVES) do
            local inst = workspace:FindFirstChild(o.path)
            local a = getAnchor(inst)
            if a then
                local d = (a.Position - eye).Magnitude
                if inRange(d) then
                    local pp = inst:FindFirstChildWhichIsA("ProximityPrompt", true)
                    local status = pp and pp.ObjectText or nil
                    local name = o.label
                    -- ObjectText carries the live requirement, e.g. "KEY NEEDED"
                    if status and status ~= "" and status:upper() ~= o.label then
                        name = o.label .. " - " .. status
                    end
                    drawTag(a, inst, label(name, d), S.ObjColor, S.ObjChams)
                end
            end
        end

        -- the Keycard hangs off the monster itself
        local m = getMonster()
        local handle = m and m:FindFirstChild("Handle")
        if handle then
            local d = (handle.Position - eye).Magnitude
            if inRange(d) then
                drawTag(handle, handle, label("KEYCARD (on Locust)", d), S.ObjColor, S.ObjChams)
            end
        end
    end

    sweepUnused()
end

-- Only accept a new ScreenGui that actually contains the window title; Roblox
-- creates unrelated CoreGui ScreenGuis all the time and picking the newest one
-- silently grabs the wrong object.
local function _holdsTitle(gui)
    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("TextLabel") and d.Text == "House of The Locust" then return true end
    end
    return false
end
for _, root in ipairs(_guiRoots) do
    for _, v in ipairs(root:GetChildren()) do
        if v:IsA("ScreenGui") and not _guiBefore[v] and _holdsTitle(v) then
            _G.__LustFluentGui = v
            break
        end
    end
    if _G.__LustFluentGui then break end
end

-- Previous version decided click-vs-drag inside InputBegan with a 4px
-- threshold. Ordinary mouse jitter between press and release exceeded that,
-- so the press was classed as a drag and the toggle was skipped - which is
-- why it took two clicks. Now the click comes from Activated (Roblox only
-- fires it for a genuine click) and a drag merely suppresses it.
local DRAG_SLOP = 10          -- px of movement before it counts as a drag

local conns = {}
local function track(c) table.insert(conns, c) return c end

local dragging, dragStart, startPos, draggedFar = false, nil, nil, false

track(btn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging, draggedFar = true, false
        dragStart = input.Position
        startPos = btn.Position
    end
end))

track(UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    if input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch then
        local delta = input.Position - dragStart
        if math.abs(delta.X) > DRAG_SLOP or math.abs(delta.Y) > DRAG_SLOP then
            draggedFar = true
        end
        if draggedFar then
            btn.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end
end))

track(UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
        -- clear the drag flag a frame later so Activated (which fires first)
        -- still sees it and can suppress itself
        task.defer(function() draggedFar = false end)
    end
end))

track(btn.Activated:Connect(function()
    if draggedFar then return end       -- that was a drag, not a click
    setHub(not hubOpen)
end))

-- Somewhere to stand that is far from the monster but still near us, so we
-- are not flung to the opposite corner of the map on every twitch.
local function dodgeTarget(myPos, monsterPos, minDist)
    local cands = {}
    local f = workspace:FindFirstChild("Cabinets")
    if f then
        for _, c in ipairs(f:GetChildren()) do
            local i = c:FindFirstChild("Interact")
            if i then table.insert(cands, i.Position + Vector3.new(0, 3, 0)) end
        end
    end
    for _, n in ipairs({ "RedDetector", "YellowDetector", "BlueDetector",
                         "ExitDoor", "SpawnLocation", "CubeRed", "CubeYellow", "CubeBlue" }) do
        local o = workspace:FindFirstChild(n)
        local a = o and (o:IsA("BasePart") and o or o:FindFirstChildWhichIsA("BasePart"))
        if a then table.insert(cands, a.Position + Vector3.new(0, 4, 0)) end
    end

    local best, bestScore
    for _, p in ipairs(cands) do
        if (p - monsterPos).Magnitude >= minDist then
            local score = (p - myPos).Magnitude          -- nearest safe spot
            if not bestScore or score < bestScore then best, bestScore = p, score end
        end
    end
    if best then return best end

    -- nothing on the list is safe: shove straight away from the monster
    local away = (myPos - monsterPos)
    if away.Magnitude < 1 then away = Vector3.new(1, 0, 0) end
    return monsterPos + away.Unit * (minDist + 20) + Vector3.new(0, 3, 0)
end

-- Distance to the nearest point ON a part, not to its centre. Detectz is a
-- 20-stud-long box, so its edge reaches ~10 studs beyond its origin - using
-- the centre badly underestimates the threat and gets you killed.
local function distToPart(part, pos)
    local lp = part.CFrame:PointToObjectSpace(pos)
    local h = part.Size * 0.5
    local c = Vector3.new(
        math.clamp(lp.X, -h.X, h.X),
        math.clamp(lp.Y, -h.Y, h.Y),
        math.clamp(lp.Z, -h.Z, h.Z))
    return (lp - c).Magnitude
end

local lastDodge = 0
local function dodgeStep()
    if not S.AutoDodge or not alive() then return end
    -- auto.lua sets this while it is deliberately standing in the Locust's
    -- blind spot to take the keycard; dodging would drag us straight out
    if _G.__LustNoDodge then return end
    local r = hrp()
    if not r then return end
    local _, root, detect = locust()
    if not root then return end

    local d = (root.Position - r.Position).Magnitude
    if detect then d = math.min(d, distToPart(detect, r.Position)) end

    -- While auto.lua is driving the character it does its own spacing via
    -- evade(). If we also dodge at the full radius the two fight each other
    -- and nothing ever completes, so back off to emergency-only.
    local threshold = S.SafeDistance
    if _G.__LustBusy then threshold = math.min(S.SafeDistance, S.EmergencyDistance) end
    if d >= threshold then return end

    -- normally rate-limit, but when it is genuinely on top of us, move NOW
    local urgent = d < S.SafeDistance * 0.5
    if not urgent and os.clock() - lastDodge < 0.2 then return end
    lastDodge = os.clock()

    local dest = dodgeTarget(r.Position, root.Position, S.SafeDistance * 2)
    r.CFrame = CFrame.new(dest)
    r.AssemblyLinearVelocity = Vector3.zero
    Stats.dodges = Stats.dodges + 1
end

local function tryRevive()
    if not S.AutoRevive then return end
    -- the game's own death screen offers Revive; use that path rather than
    -- inventing one. Fall back to the remote if the button is not present.
    local gui = LocalPlayer:FindFirstChild("PlayerGui")

        for _, v in ipairs(gui:GetDescendants()) do
            if (v:IsA("TextButton") or v:IsA("ImageButton")) and v.Visible then
                local t = (v:IsA("TextButton") and v.Text or v.Name):lower()
                if t:find("revive") then
                    pcall(function()
                        for _, c in ipairs(getconnections(v.MouseButton1Click)) do c:Fire() end
                    end)
                    clicked = true
                    break
                end
            end
        end
    end
    if not clicked then
        local ev = RS:FindFirstChild("ReviveEvent")
        if ev then pcall(function() ev:FireServer() end) end
    end
    Stats.revives = Stats.revives + 1
end

-- hold>0 prompts (the cabinets).

--     InputHoldEnd() is mandatory - waiting past HoldDuration alone does nothing.

--      silently cancelled all 28 searches.)
local FleeRadius = 45      -- bail out of a hold if the Locust gets this close

-- Walk a ring of offsets until the engine actually SHOWS the prompt.
-- A fixed "+LookVector * 2.5" lands inside geometry on some cabinets, and a
-- prompt that never shows can never be triggered - that silently ate 8 of the
-- 28 cabinets on the previous run.
local function approach(part, prompt)
    prompt.RequiresLineOfSight = false          -- client-side gate only
    local shown = false
    local c = PPS.PromptShown:Connect(function(p) if p == prompt then shown = true end end)

    local offsets = {
        part.CFrame.LookVector * 2.5,
        part.CFrame.LookVector * -2.5,
        part.CFrame.RightVector * 2.5,
        part.CFrame.RightVector * -2.5,
        Vector3.new(0, 2.5, 0),
    }
    for _, off in ipairs(offsets) do
        tp(CFrame.new(part.Position + off + Vector3.new(0, 0.5, 0), part.Position))
        local t = 0
        while t < 0.8 do
            task.wait(0.1); t = t + 0.1
            if shown then c:Disconnect() return true end
        end
    end
    c:Disconnect()
    return shown
end

local function locustPos()
    local e = workspace:FindFirstChild("Entities")
    local l = e and e:FindFirstChild("Locust")
    local a = l and (l.PrimaryPart or l:FindFirstChild("HumanoidRootPart"))
    return a and a.Position or nil
end

-- The spot on the map currently farthest from the Locust, used as a bolt-hole.
local function safeSpot()
    local mp = locustPos()
    if not mp then return nil end
    local cands = {}
    local f = workspace:FindFirstChild("Cabinets")
    if f then
        for _, c in ipairs(f:GetChildren()) do
            local i = c:FindFirstChild("Interact")
            if i then table.insert(cands, i.Position + Vector3.new(0, 3, 0)) end
        end
    end
    for _, n in ipairs({ "RedDetector", "YellowDetector", "BlueDetector", "ExitDoor", "SpawnLocation" }) do
        local o = workspace:FindFirstChild(n)
        local a = o and (o:IsA("BasePart") and o or o:FindFirstChildWhichIsA("BasePart"))
        if a then table.insert(cands, a.Position + Vector3.new(0, 4, 0)) end
    end
    local best, bd
    for _, p in ipairs(cands) do
        local d = (p - mp).Magnitude
        if not bd or d > bd then best, bd = p, d end
    end
    return best, bd
end

-- Before working at a spot: if the Locust is near it, run to the far side of
-- the map and wait there until it wanders off, then come back.
local function evade(targetPos, minDist)
    minDist = minDist or (FleeRadius * 2)
    local hidden, waited = false, 0
    while waited < 40 do
        if Abort then return false end
        local mp = locustPos()
        if not mp then return true end
        if (mp - targetPos).Magnitude >= minDist then
            if hidden then say("Locust clear - back to work") end
            return true
        end
        local spot = safeSpot()
        if spot then tp(CFrame.new(spot)) end
        if not hidden then
            say("Locust %dm from target - hiding", math.floor((mp - targetPos).Magnitude))
            hidden = true
        end
        task.wait(1)
        waited = waited + 1
    end
    return false
end

local function holdPrompt(part, p, seconds)
    if not (part and p) then return false end
    if not approach(part, p) then return false end   -- prompt never activated
    if not alive() then return false end

    local done = false
    local c = p.Triggered:Connect(function() done = true end)

    if InstantSearch then
        -- HoldDuration is a client-side gate: zero it and the trigger fires at
        -- once, so we stand still for ~0.3s instead of 5.3s. Far less exposure.
        p.HoldDuration = 0
        task.wait(0.2)
        pcall(function() p:InputHoldBegin() end)
        pcall(function() p:InputHoldEnd() end)
    else
        pcall(function() p:InputHoldBegin() end)
        local t = 0
        while t < (seconds or 5.3) do        -- do NOT move in here
            task.wait(0.1); t = t + 0.1
            local mp = locustPos()
            local r = hrp()
            local tooClose = mp and r and (mp - r.Position).Magnitude < FleeRadius
            if Abort or not alive() or tooClose then
                pcall(function() p:InputHoldEnd() end)
                c:Disconnect()
                if tooClose then say("Locust closing in - aborting this cabinet") end
                return false
            end
        end
        pcall(function() p:InputHoldEnd() end)
    end

    local w = 0
    while not done and w < 1.2 do task.wait(0.1); w = w + 0.1 end
    c:Disconnect()
    return done
end

-- The lock prompts keep saying "KEY NEEDED" / "KEYCARD NEEDED" even once you
-- are carrying the item - ObjectText is static flavour text, not state. What
-- actually tells you is a Tool of that name in the backpack or equipped.
local function hasTool(name)
    local c = char()
    if c and c:FindFirstChild(name) then return true end
    return LocalPlayer.Backpack:FindFirstChild(name) ~= nil
end

local function equipTool(name)
    local c = char()
    local h = c and c:FindFirstChildOfClass("Humanoid")
    local t = LocalPlayer.Backpack:FindFirstChild(name)
    if t and h then pcall(function() h:EquipTool(t) end); task.wait(0.4); return true end
    return c and c:FindFirstChild(name) ~= nil
end

-- Reading lock progress is genuinely awkward here:
--   * ObjectText is static flavour text ("KEY NEEDED" forever), useless.
--   * prompt.Enabled going false is a TRANSIENT cooldown while the server
--     chews on an interaction - it flips back to true a few seconds later.
--     It means neither success nor "burned", so never treat it as either;
--     just wait for it to come back before firing again.
-- The only trustworthy success signal is the item being consumed.
local Unlocked = {}
local LOCKS = { "Keyhole", "KeycardReader", "DoorWire" }
local UNLOCK_SETTLE = 8   -- seconds disabled before we call it genuinely done

-- true while the lock's prompt is disabled (cooldown OR done - see below)
local function lockBusy(name)
    local o = workspace:FindFirstChild(name)
    if not o then return false end
    local p = o:FindFirstChildWhichIsA("ProximityPrompt", true)
    return p ~= nil and not p.Enabled
end

-- Watch when each lock was last seen ENABLED. A failed attempt disables the
-- prompt for a few seconds and then it comes back; a real unlock disables it
-- for good. So "disabled continuously for UNLOCK_SETTLE seconds" is the only
-- reliable done signal - the tool is NOT consumed, you keep it.
-- Seed every lock at load time. Without this, a lock that is ALREADY disabled
-- when the script starts never gets a timestamp and can never settle, so it
-- reads as "on cooldown" forever.
local LastEnabled = {}
for _, n in ipairs(LOCKS) do LastEnabled[n] = os.clock() end

task.spawn(function()
    while not Fluent.Unloaded do
        for _, n in ipairs(LOCKS) do
            if not lockBusy(n) then LastEnabled[n] = os.clock() end
        end
        task.wait(0.5)
    end
end)

local function lockSettled(name)
    if workspace:FindFirstChild(name) == nil then return true end
    local last = LastEnabled[name]
    return last ~= nil and lockBusy(name) and (os.clock() - last) >= UNLOCK_SETTLE
end

local function waitLockReady(name, timeout)
    local t = 0
    while lockBusy(name) and t < (timeout or 10) do
        task.wait(0.5); t = t + 0.5
        if Abort or lockSettled(name) then return false end
    end
    return not lockBusy(name)
end

local function lockDone(name)
    if Unlocked[name] then return true end
    if workspace:FindFirstChild(name) == nil then return true end
    if lockSettled(name) then Unlocked[name] = true; return true end
    return false
end

local function doCubes()
    if cubesDone() then say("cube puzzle already solved"); return true end
    for _, e in ipairs(CUBES) do
        if Abort then return false end
        local cube, det = workspace:FindFirstChild(e[1]), workspace:FindFirstChild(e[2])
        if cube and det then
            say("moving %s", e[1])
            evade(cube.Position)
            tp(CFrame.new(cube.Position + Vector3.new(0, 4, 6)))
            task.wait(1.2)
            for _ = 1, 80 do
                if Abort then return false end
                local a = cube.Position
                local b = det.Position + Vector3.new(0, 3, 0)
                if (b - a).Magnitude < 2 then break end
                local nextPos = a + (b - a).Unit * math.min(25, (b - a).Magnitude)
                cube.CFrame = CFrame.new(nextPos)
                cube.AssemblyLinearVelocity = Vector3.zero
                tp(CFrame.new(nextPos + Vector3.new(0, 4, 8)))
                task.wait(0.1)
            end
            task.wait(0.5)
        end
    end
    task.wait(2)
    say(cubesDone() and "cube puzzle solved - ColorDoor open" or "cube puzzle incomplete")
    return cubesDone()
end

--------------------------------------------------- STEP 3: cabinets -> key
local function doCabinets()
    if haveKey() then say("already have the key"); return true end
    local folder = workspace:FindFirstChild("Cabinets")
    if not folder then return false end

    local function remaining()
        local t = {}
        for _, c in ipairs(folder:GetChildren()) do
            local i = c:FindFirstChild("Interact")
            local p = i and i:FindFirstChildOfClass("ProximityPrompt")
            if p and p.Enabled and not Searched[c] then
                table.insert(t, { model = c, part = i, prompt = p })
            end
        end
        return t
    end

    -- Each cabinet needs 5.3s of standing perfectly still, so always take the
    -- one FARTHEST from the Locust right now rather than a fixed route.
    local total = #remaining()
    local failures = {}
    for n = 1, total * 6 do
        if Abort then return false end
        -- stop the moment we have it, the rest of the cabinets do not matter
        if haveKey() then
            say("KEY FOUND - stopping search, %d cabinets left unopened", #remaining())
            return true
        end
        local list = remaining()
        if #list == 0 then break end
        if not alive() then
            say("died - waiting to respawn")
            if not waitAlive(25) then return false end
        end

        local mp = locustPos()
        local pick, best
        for _, e in ipairs(list) do
            if (failures[e.model] or 0) < 4 then
                local score = mp and (e.part.Position - mp).Magnitude or 0
                if not best or score > best then pick, best = e, score end
            end
        end
        -- everything left has failed a lot: clear the strikes and go round again
        if not pick then
            if next(failures) == nil then break end
            failures = {}
            say("retrying the cabinets that kept failing")
            pick, best = list[1], 0
        end

        -- wait out the Locust rather than dying at this cabinet
        if not evade(pick.part.Position) then
            failures[pick.model] = (failures[pick.model] or 0) + 1
        else
            local ok = holdPrompt(pick.part, pick.prompt, 5.3)
            if ok then
                Searched[pick.model] = true
                say("searched %d/%d (Locust was %dm away)", total - #remaining() + 1, total, math.floor(best or 0))
            else
                failures[pick.model] = (failures[pick.model] or 0) + 1
            end
        end
    end
    if haveKey() then return true end
    say("no key after searching everything reachable")
    return false
end

--------------------------------------------------- STEP 4: keycard off the Locust
local function doKeycard()
    if haveKeycard() then say("already have the keycard"); return true end
    local ent = workspace:FindFirstChild("Entities")
    local locust = ent and ent:FindFirstChild("Locust")
    local handle = locust and locust:FindFirstChild("Handle")
    if not handle then say("locust handle not found"); return false end
    local p = handle:FindFirstChildOfClass("ProximityPrompt")
    if not p then say("no keycard prompt"); return false end
    local root = locust.PrimaryPart or locust:FindFirstChild("HumanoidRootPart")
    local dz = locust:FindFirstChild("Detectz")
    if not root then say("locust root missing"); return false end

    -- Detectz is a 20-stud-WIDE bar at head height (local Y +3.6, 2.4 tall),
    -- centred on the monster - a sideways sweep, not a forward reach. The
    -- Handle dangles in front at local Z +2.9. That leaves a pocket in front
    -- of and beside the monster that is inside prompt range but outside the
    -- bar. Ride that pocket instead of teleporting onto its face.
    local POCKETS = {
        Vector3.new(6, 0, 6), Vector3.new(4, 0, 8),
        Vector3.new(-6, 0, 6), Vector3.new(6, 2, 6),
    }

    local function boxClearance(pos)
        if not dz then return 99 end
        local l = dz.CFrame:PointToObjectSpace(pos)
        local hs = dz.Size * 0.5
        local c = Vector3.new(math.clamp(l.X,-hs.X,hs.X), math.clamp(l.Y,-hs.Y,hs.Y), math.clamp(l.Z,-hs.Z,hs.Z))
        return (l - c).Magnitude
    end

    -- stop anti.lua yanking us out of the pocket mid-grab
    _G.__LustNoDodge = true
    local home = hrp() and hrp().CFrame
    say("grabbing keycard from the Locust's blind spot")

    local got = false
    for attempt = 1, #POCKETS * 3 do
        if Abort or not alive() then break end
        local pocket = POCKETS[((attempt - 1) % #POCKETS) + 1]

        -- hold the pocket for a moment, re-solving it every frame because the
        -- monster keeps turning; bail instantly if the bar sweeps onto us
        local t = 0
        while t < 1.2 do
            if Abort or not alive() then break end
            local r = hrp()
            if not r then break end
            local dest = root.CFrame:PointToWorldSpace(pocket)
            if boxClearance(dest) < 3 then break end        -- pocket collapsed
            r.CFrame = CFrame.new(dest, handle.Position)
            r.AssemblyLinearVelocity = Vector3.zero
            pcall(fireproximityprompt, p)
            if p.Parent == nil or haveKeycard() then got = true break end
            task.wait(0.1); t = t + 0.1
        end
        if got then say("keycard grabbed (attempt %d)", attempt) break end
    end

    _G.__LustNoDodge = false
    if home and alive() then pcall(tp, home) end
    task.wait(0.8)
    got = got or haveKeycard() or (p.Parent == nil)
    say(got and "keycard acquired" or "keycard grab failed")
    return got
end

--------------------------------------------------- STEP 5: the three locks
local function doLocks()
    -- each lock wants its item actually equipped, not just carried
    local order = {
        { "Keyhole",       "keyhole",        "Key"        },
        { "KeycardReader", "keycard reader", "Keycard"    },
        { "DoorWire",      "door wire",      "WireCutter" },
    }
    for _, e in ipairs(order) do
        if Abort then return false end
        local name, label, toolName = e[1], e[2], e[3]

        if lockDone(name) then
            say("%s already done", label)
        elseif not hasTool(toolName) then
            say("skipping %s - no %s yet", label, toolName)
        else
            -- retryable: the prompt is not one-shot, it just goes on cooldown
            for attempt = 1, 4 do
                if Abort or lockDone(name) then break end
                local o = workspace:FindFirstChild(name)
                local p = promptOf(o)
                local a = anchorOf(o)
                if not (o and p and a) then break end

                if a then evade(a.Position) end
                if not waitLockReady(name, 8) then
                    say("%s still on cooldown, waiting", label)
                end
                if not alive() then
                    if not waitAlive(25) then return false end
                end
                -- the tool must be EQUIPPED, not just carried, and it has to
                -- stay equipped through the teleporting that approach() does
                if not hasTool(toolName) then
                    say("lost the %s (died?) - cannot do %s", toolName, label)
                    break
                end
                equipTool(toolName)

                -- All three locks are E / OnePerButton and sit ~18 studs apart,
                -- so only the nearest is live. approach() walks offsets until
                -- the engine reports this exact prompt as shown.
                if not approach(a, p) then
                    say("%s attempt %d - prompt would not show", label, attempt)
                else
                    equipTool(toolName)          -- re-equip after the hops
                    task.wait(0.3)
                    pcall(fireproximityprompt, p)
                    -- give it longer than the cooldown so a failed attempt's
                    -- prompt has time to come back before we judge it
                    local waited = 0
                    while waited < UNLOCK_SETTLE + 2 do
                        task.wait(0.5); waited = waited + 0.5
                        if lockDone(name) then break end
                        if not lockBusy(name) then break end   -- came back = failed
                    end
                    if lockDone(name) then
                        say("%s UNLOCKED", label)
                        break
                    end
                    say("%s attempt %d - prompt came back, did not take", label, attempt)
                end
            end
            if not lockDone(name) then say("%s did NOT unlock", label) end
        end
    end
    return true
end

end

local function runFrom(first)
    if Running then say("already running"); return end
    Running, Abort = true, false
    HomeCF = hrp() and hrp().CFrame or nil

    -- Tell anti.lua we are driving the character. Its Auto-Dodge drops to an
    -- emergency-only threshold while this is set; otherwise both scripts
    -- teleport the player every frame and neither ever converges.
    _G.__LustBusy = true

    task.spawn(function()
        for i = first or 1, #Steps do
            if Abort then break end
            local s = Steps[i]
            if s.risky and not RiskKeycard then
                say("skipping %s (enable 'Allow lethal keycard grab')", s.name)
                continue
            end
            say("STEP %d/%d - %s", i, #Steps, s.name)
            local ok, err = pcall(s.fn)
            if not ok then warn("[AutoEscape] " .. s.name .. ": " .. tostring(err)) end
            task.wait(0.4)
        end
        if Abort and HomeCF then pcall(tp, HomeCF) end
        say(Abort and "aborted" or "sequence finished")
        Running = false
        _G.__LustBusy = false
    end)
end

end

    end
    return false
end
