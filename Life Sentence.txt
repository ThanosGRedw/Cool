    if isfile then
        if readfile(argument) == nil then
            resetdefault(argument, "nil", 20)
        end
        local readfileResult = readfile(argument)
        local split = readfileResult:split(",")
        local text = ""
        for key, item in split do
            if key == secondaryArgument then
                text = tostring(text .. secondaryText .. ",")

        writefile(argument, text)
    end

        local function integerToBytes(value, byteCount)
            local integerToBytesText = ""
            for index = 1, byteCount do
                local byteValue = value % 256
                integerToBytesText = string.char(byteValue) .. integerToBytesText

            end

        local function bytesToInteger(bytes, offset)

            local difference = 64 - ((messageLength + 9) % 64)
            messageLength = integerToBytes(8 * messageLength, 8)

            assert((#message % 64) == 0)
            return message
        end

        local function secondaryHandler(bytes, number, hashState)
            local data = {}
            for index = 1, 16 do
                data[index] = bytesToInteger(bytes, number + ((index - 1) * 4))
            end
            for index = 17, 64 do
                local leftValue = data[index - 15]
                local xorResult = bitXorMany(
                    alternateHandleNumber(leftValue, 7),
                    alternateHandleNumber(leftValue, 18),
                    handler(leftValue, 3)

                leftValue = data[index - 2]
                data[index] = (
                    data[index - 16]
                    + xorResult
                    + data[index - 7]
                    + bitXorMany(
                        alternateHandleNumber(leftValue, 17),
                        alternateHandleNumber(leftValue, 19),
                        handler(leftValue, 10)
                    )
                ) % power
            end

                hashState[1],
                hashState[2],
                hashState[3],
                hashState[4],
                hashState[5],
                hashState[6],
                hashState[7],
                hashState[8]

                local xorResult = bitXorMany(
                    alternateHandleNumber(leftValue, 2),
                    alternateHandleNumber(leftValue, 13),
                    alternateHandleNumber(leftValue, 22)
                )
                local secondaryXorResult = bitXorMany(
                    bitAnd(leftValue, rightValue),
                    bitAnd(leftValue, secondaryRightValue),
                    bitAnd(rightValue, secondaryRightValue)
                )
                local number = (xorResult + secondaryXorResult) % power
                local alternateXorResult = bitXorMany(
                    alternateHandleNumber(remainder, 6),
                    alternateHandleNumber(remainder, 11),
                    alternateHandleNumber(remainder, 25)
                )
                local additionalXorResult = bitXorMany(
                    bitAnd(remainder, additionalRightValue),
                    bitAnd(handleNumber(remainder), fallbackRightValue)
                )
                local secondaryRemainder = (
                    nestedRightValue
                    + alternateXorResult
                    + additionalXorResult
                    + remainderData[index]
                    + data[index]
                ) % power
                nestedRightValue = fallbackRightValue
                fallbackRightValue = additionalRightValue
                additionalRightValue = remainder
                remainder = (alternateRightValue + secondaryRemainder) % power
                alternateRightValue = secondaryRightValue
                secondaryRightValue = rightValue
                rightValue = leftValue
                leftValue = (secondaryRemainder + number) % power
            end
            hashState[1] = (hashState[1] + leftValue) % power
            hashState[2] = (hashState[2] + rightValue) % power
            hashState[3] = (hashState[3] + secondaryRightValue) % power
            hashState[4] = (hashState[4] + alternateRightValue) % power
            hashState[5] = (hashState[5] + remainder) % power
            hashState[6] = (hashState[6] + additionalRightValue) % power
            hashState[7] = (hashState[7] + fallbackRightValue) % power

                    + Vector3.new(0, 5, 0)
                local humanoid = localValuePlayer.Character.Humanoid
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                task.wait(0.1)
                local updateCFrameNumber = 0
                repeat
                    handleCFrame(promptPartContainer.PromptPart:WaitForChild("ProximityPrompt"))

                    task.wait(0.1)

            end
        end
    end
