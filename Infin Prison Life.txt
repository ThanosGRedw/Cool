local warned = {}
local function safe(name, fn, ...)
	local ok, err = pcall(fn, ...)
	if not ok and not warned[name] then
		warned[name] = true
		warn("[INFIN] " .. name .. ": " .. tostring(err))
	end
	return ok
end

local function hasTool(name)
	local bp = LP:FindFirstChildOfClass("Backpack")

local function equipTool(name)
	local h = getHum()
	local c = LP.Character
	if not h or (c and c:FindFirstChild(name)) then return end
	local bp = LP:FindFirstChildOfClass("Backpack")
	local tool = bp and bp:FindFirstChild(name)
	if tool then h:EquipTool(tool) end
end
local function teleportTo(target)
	local r, h = getRoot(), getHum()
	if not r then return end
	if h and h.Sit then h.Sit = false end
	if typeof(target) == "Vector3" then target = CFrame.new(target + Vector3.new(0, 3, 0)) end
	r.CFrame = target
end

local function new(class, props)
	local o = Instance.new(class)
	if o:IsA("GuiObject") then o.BorderSizePixel = 0 end
	local parent
	for k, v in pairs(props or {}) do
		if k == "Parent" then parent = v else o[k] = v end
	end
	if parent then o.Parent = parent end
	return o
end
local function corner(p, r)
	local radius = (typeof(r) == "UDim" and r) or UDim.new(0, math.floor((r or 6) * 1.6 + 0.5))
	return new("UICorner", { CornerRadius = radius, Parent = p })
end
local function stroke(p, col, th, tr)
	return new("UIStroke", {
		Color = col or C.stroke, Thickness = th or 1, Transparency = tr or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = p,
	})
end
local function padding(p, t, r, b, l)
	return new("UIPadding", {
		PaddingTop = UDim.new(0, t), PaddingRight = UDim.new(0, r or t),
		PaddingBottom = UDim.new(0, b or t), PaddingLeft = UDim.new(0, l or r or t), Parent = p,
	})
end
local function list(p, gap, dir, halign, valign)
	return new("UIListLayout", {
		Padding = UDim.new(0, gap or 6), FillDirection = dir or Enum.FillDirection.Vertical,
		SortOrder = Enum.SortOrder.LayoutOrder,
		HorizontalAlignment = halign or Enum.HorizontalAlignment.Left,
		VerticalAlignment = valign or Enum.VerticalAlignment.Top, Parent = p,
	})
end
local function tween(o, props, t, style)
	local tw = TweenService:Create(o, TweenInfo.new(t or 0.18, style or Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
	tw:Play()
	return tw
end
local function themed(obj, prop, key)
	table.insert(Infin.Themed, { obj, prop, key })
	obj[prop] = Theme[key]
end
local function onTheme(fn)
	table.insert(Infin.ThemeHooks, fn)
	fn()
end
local function applyTheme(name)
	Theme = THEMES[name] or THEMES.Midnight
	for _, t in ipairs(Infin.Themed) do
		pcall(function() t[1][t[2]] = Theme[t[3]] end)
	end
	for _, fn in ipairs(Infin.ThemeHooks) do pcall(fn) end
end
local function drawLine(f, a, b, thick)
	local d = b - a
	f.Size = UDim2.fromOffset(d.Magnitude, thick)
	f.Position = UDim2.fromOffset((a.X + b.X) / 2, (a.Y + b.Y) / 2)
	f.Rotation = math.deg(math.atan2(d.Y, d.X))
	f.Visible = true
end

local function makeDraggable(handle, target, onClick)
	local dragging, moved, startInput, startPos = false, false, nil, nil
	handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging, moved, startInput, startPos = true, false, i.Position, target.Position
			local conn
			conn = i.Changed:Connect(function()
				if i.UserInputState == Enum.UserInputState.End then
					dragging = false
					conn:Disconnect()
					if not moved and onClick then onClick() end
				end
			end)
		end
	end)
	bind(UIS.InputChanged, function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local d = i.Position - startInput
			if d.Magnitude > 5 then moved = true end
			target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
end
makeDraggable(header, main)

		Position = UDim2.new(0, 4, 0, 18), Size = UDim2.new(1, -8, 0, 22),
		BackgroundTransparency = 1, Text = "", ZIndex = 7, Parent = row,
	})

	local decimals = (step < 1) and ((step < 0.1) and 2 or 1) or 0
	local function fmt(v)
		if decimals == 0 then return tostring(math.floor(v + 0.5)) end
		return string.format("%." .. decimals .. "f", v)
	end
	local function set(v)
		v = tonumber(v) or default
		v = math.clamp(math.floor((v - min) / step + 0.5) * step + min, min, max)
		v = tonumber(fmt(v))
		F[flag] = v
		local a = (max == min) and 0 or (v - min) / (max - min)
		fill.Size = UDim2.fromScale(a, 1)
		knob.Position = UDim2.fromScale(a, 0.5)
		valLbl.Text = fmt(v) .. suffix
		if cb then safe("slider " .. flag, cb, v) end
	end
	local dragging = false
	local function fromX(x)
		local a = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
		set(min + (max - min) * a)
	end
	hit.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			fromX(i.Position.X)
		end
	end)
	bind(UIS.InputChanged, function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			fromX(i.Position.X)
		end
	end)
	bind(UIS.InputEnded, function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	Infin.Setters[flag] = set
	Infin.Defaults[flag] = default
	UI.register(sec, row, text)
	set(default)
	return set
end

function UI.Selector(sec, text, flag, options, default, cb)
	local row = UI.rowBase(sec, 30, false)
	UI.rowLabel(row, text, 150)
	local box = new("Frame", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -6, 0.5, 0), Size = UDim2.fromOffset(134, 22),
		BackgroundColor3 = C.bg, BackgroundTransparency = 0.2, ZIndex = 4, Parent = row,
	})
	corner(box, 5)
	stroke(box)
	local function arrow(txt, x, ax)
		local b = new("TextButton", {
			AnchorPoint = Vector2.new(ax, 0.5), Position = UDim2.new(x, 0, 0.5, 0), Size = UDim2.fromOffset(22, 22),
			BackgroundTransparency = 1, Text = txt, Font = Enum.Font.GothamBold, TextSize = 13, ZIndex = 5, Parent = box,
		})
		themed(b, "TextColor3", "accent")
		return b
	end
	local prev, nxt = arrow("<", 0, 0), arrow(">", 1, 1)
	local lbl = new("TextLabel", {
		Position = UDim2.fromOffset(22, 0), Size = UDim2.new(1, -44, 1, 0), BackgroundTransparency = 1,
		Font = Enum.Font.GothamSemibold, TextSize = 11, TextColor3 = C.text,
		TextTruncate = Enum.TextTruncate.AtEnd, ZIndex = 5, Parent = box,
	})
	local value
	local function getOpts()
		if type(options) == "function" then return options() end
		return options
	end
	local function set(v)
		value = v
		if flag then F[flag] = v end
		lbl.Text = (v ~= nil) and tostring(v) or "none"
		if cb then safe("selector " .. tostring(flag or text), cb, v) end
	end
	local function step(dir)
		local opts = getOpts()
		if #opts == 0 then set(nil) return end
		local idx = table.find(opts, value)
		if not idx then
			idx = (dir == 1) and 1 or #opts
		else
			idx = ((idx - 1 + dir) % #opts) + 1
		end
		set(opts[idx])
	end
	prev.MouseButton1Click:Connect(function() step(-1) end)
	nxt.MouseButton1Click:Connect(function() step(1) end)
	if flag then
		Infin.Setters[flag] = set
		Infin.Defaults[flag] = default
	end
	UI.register(sec, row, text)
	set(default)
	return set, function() return value end
end

function UI.runSearch()
	local q = string.lower(searchBox.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
	if q == "" then
		for _, e in ipairs(Infin.Search) do e.frame.Visible = true end
		for _, s in ipairs(Infin.Sections) do s.card.Visible = true end
		return
	end
	local firstTab, currentHas = nil, false
	for _, e in ipairs(Infin.Search) do
		local m = string.find(e.text, q, 1, true) ~= nil
		e.frame.Visible = m
		if m then
			firstTab = firstTab or e.sec.tab
			if e.sec.tab == currentTab then currentHas = true end
		end
	end
	for _, s in ipairs(Infin.Sections) do
		local any = false
		for _, e in ipairs(s.items) do
			if e.frame.Visible then any = true break end
		end
		s.card.Visible = any
	end
	if firstTab and not currentHas then UI.selectTab(firstTab) end
end
bind(searchBox:GetPropertyChangedSignal("Text"), UI.runSearch)

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function isVisible(part, char)
	local origin = Camera.CFrame.Position
	rayParams.FilterDescendantsInstances = { LP.Character, Camera }
	local res = workspace:Raycast(origin, part.Position - origin, rayParams)
	return res == nil or res.Instance:IsDescendantOf(char)
end

local function validTarget(p, teamCheck)
	if p == LP then return false end
	local c = p.Character
	local h = c and c:FindFirstChildOfClass("Humanoid")
	if not h or h.Health <= 0 or not c:FindFirstChild("HumanoidRootPart") then return false end
	if teamCheck and p.Team ~= nil and p.Team == LP.Team then return false end
	if F.aim_ff and c:FindFirstChildOfClass("ForceField") then return false end
	return true
end

local function candidateParts(c, mode)
	local out = {}
	local head = c:FindFirstChild("Head")
	local torso = c:FindFirstChild("Torso") or c:FindFirstChild("UpperTorso") or c:FindFirstChild("HumanoidRootPart")
	if mode == "Head" then
		local p = head or torso
		if p then table.insert(out, p) end
	elseif mode == "Torso" then
		if torso then table.insert(out, torso) end
	else
		if head then table.insert(out, head) end
		if torso then table.insert(out, torso) end
	end
	return out
end

local function aimOrigin()
	local mode = F.aim_origin or "Auto"
	if mode == "Center" or (mode == "Auto" and UIS.TouchEnabled and not UIS.MouseEnabled) then
		local vp = Camera.ViewportSize
		return Vector2.new(vp.X / 2, vp.Y / 2)
	end
	return UIS:GetMouseLocation()
end

local aimTargetName

local function forcedTarget()
	if not F.tgt_only or not aimTargetName then return nil end
	return Players:FindFirstChild(aimTargetName)
end

local function findTarget(fov, partMode, teamCheck, wallCheck, maxDist)
	local origin = aimOrigin()
	local camPos = Camera.CFrame.Position
	local forced = forcedTarget()
	local free = forced ~= nil and F.tgt_ignorefov
	local bestP, bestPart, bestD = nil, nil, fov
	for _, p in ipairs(Players:GetPlayers()) do
		if (not forced or p == forced) and validTarget(p, teamCheck and not forced) then
			local c = p.Character
			for _, part in ipairs(candidateParts(c, partMode)) do
				if free or (part.Position - camPos).Magnitude <= (maxDist or 1e9) then
					local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
					local d = (Vector2.new(sp.X, sp.Y) - origin).Magnitude
					local inFov = onScreen and d < bestD
					local pick = free and (bestPart == nil or d < bestD)
					if (inFov or pick) and (not wallCheck or isVisible(part, c)) then
						bestP, bestPart, bestD = p, part, d
					end
				end
			end
		end
	end
	return bestP, bestPart
end

local function closestToCrosshair()
	local origin = aimOrigin()
	local best, bestD = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		local r = getRoot(p)
		if r and validTarget(p, false) then
			local sp, onScreen = Camera:WorldToViewportPoint(r.Position)
			local d = (Vector2.new(sp.X, sp.Y) - origin).Magnitude
			if onScreen and d < bestD then best, bestD = p, d end
		end
	end
	return best
end

local function predict(part, amount)
	return part.Position + part.AssemblyLinearVelocity * (amount or 0)
end

local lockTarget, lockPart

local function lockHeld()
	if mobileAimOn and F.aim_mobilebtn then return true end
	local k = F.aim_key
	if k == "Always" then return true end
	if k == "Right Mouse" then return UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) end
	if k == "Left Mouse" then return UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) end
	local ok, kc = pcall(function() return Enum.KeyCode[k] end)
	return (ok and kc and UIS:IsKeyDown(kc)) or false
end

local function aimStep(dt)
	if not F.aim_enabled or not lockHeld() or UIS:GetFocusedTextBox() then
		lockTarget, lockPart = nil, nil
		return
	end
	local keep = F.aim_sticky and lockTarget and lockPart and lockPart.Parent and validTarget(lockTarget, F.aim_team)
	if not keep then
		lockTarget, lockPart = findTarget(F.aim_fov, F.aim_part, F.aim_team, F.aim_wall, F.aim_dist)
	end
	if lockPart then
		local goal = CFrame.lookAt(Camera.CFrame.Position, predict(lockPart, F.aim_pred))
		local smooth = math.max(F.aim_smooth or 1, 1)
		local alpha = 1 - (1 - 1 / smooth) ^ (dt * 60)
		Camera.CFrame = Camera.CFrame:Lerp(goal, math.clamp(alpha, 0, 1))
	end
end

local hooked = false
local function setupHooks()
	if hooked then return true end
	if not (hookmetamethod and getnamecallmethod and checkcaller) then return false end
	local wrap = newcclosure or function(f) return f end
	local RAY_METHODS = {
		Raycast = true, FindPartOnRay = true,
		FindPartOnRayWithIgnoreList = true, FindPartOnRayWithWhitelist = true,
	}

	local oldNamecall
	oldNamecall = hookmetamethod(game, "__namecall", wrap(function(self, ...)
		local target = Infin.SilentPos
		if target and not Infin.Unloaded and self == workspace and not checkcaller() then
			local method = getnamecallmethod()
			if RAY_METHODS[method] and math.random(1, 100) <= (Infin.Flags.silent_hit or 100) then
				local args = table.pack(...)
				if method == "Raycast" then
					local origin, dir = args[1], args[2]
					if typeof(origin) == "Vector3" and typeof(dir) == "Vector3" then
						local to = target - origin
						args[2] = to.Unit * math.max(dir.Magnitude, to.Magnitude + 5)
					end
				else
					local ray = args[1]
					if typeof(ray) == "Ray" then
						local to = target - ray.Origin
						args[1] = Ray.new(ray.Origin, to.Unit * math.max(ray.Direction.Magnitude, to.Magnitude + 5))
					end
				end
				return oldNamecall(self, table.unpack(args, 1, args.n))
			end
		end
		return oldNamecall(self, ...)
	end))

	local oldIndex
	oldIndex = hookmetamethod(game, "__index", wrap(function(self, key)
		local target = Infin.SilentPos
		if target and self == Mouse and not Infin.Unloaded and not checkcaller() then
			if key == "Hit" then return CFrame.new(target) end
			if key == "Target" then return Infin.SilentPart end
		end
		return oldIndex(self, key)
	end))

	hooked = true
	return true
end

Guns.giverCache = {}

function Guns.findPickup(name)
	local cached = Guns.giverCache[name]
	if cached and cached.Parent then return cached end
	local items = workspace:FindFirstChild("Prison_ITEMS")
	if not items then return nil end
	for _, folderName in ipairs({ "giver", "single" }) do
		local folder = items:FindFirstChild(folderName)
		local model = folder and folder:FindFirstChild(name)
		if model then
			local pickup = model:FindFirstChild("ITEMPICKUP")
				or (model:IsA("BasePart") and model)
				or model:FindFirstChildWhichIsA("BasePart", true)
			if pickup then
				Guns.giverCache[name] = pickup
				return pickup
			end
		end
	end
	return nil
end

function Guns.standOnGiver(name, pickup)
	local model = pickup.Parent
	local prompt = model and model:FindFirstChildWhichIsA("ProximityPrompt", true)
	local handler = getRemote("ItemHandler")
	local standCF = CFrame.new(pickup.Position + Vector3.new(0, pickup.Size.Y / 2 + 2.6, 0))
	local t0, tick = os.clock(), 0
	while os.clock() - t0 < 1.8 do
		local r = getRoot()
		if not r then return false end
		r.CFrame = standCF
		r.AssemblyLinearVelocity = Vector3.zero
		if firetouchinterest then
			pcall(firetouchinterest, r, pickup, 0)
			pcall(firetouchinterest, r, pickup, 1)
		end
		if prompt and fireproximityprompt then pcall(fireproximityprompt, prompt) end
		if handler and tick % 3 == 0 then
			task.spawn(pcall, handler.InvokeServer, handler, pickup)
		end
		tick += 1
		task.wait(0.12)
		if hasTool(name) then return true end
	end
	return hasTool(name) ~= nil
end

function Guns.giveItem(name, quiet, noReturn)
	if hasTool(name) then return true end
	local pickup = Guns.findPickup(name)
	if not pickup then
		if not quiet then notify(name .. " giver not found") end
		return false
	end
	local r = getRoot()
	if not r then return false end
	local back = r.CFrame
	local got = Guns.standOnGiver(name, pickup)
	if not noReturn then
		r = getRoot()
		if r then r.CFrame = back end
	end
	if not got and not quiet then notify("Couldn't get " .. name .. " (team locked?)") end
	return got
end

Guns.gunOriginals = setmetatable({}, { __mode = "k" })

function Guns.modGun(tool)
	if not (tool and tool:IsA("Tool")) then return end
	local states
	local mod = tool:FindFirstChild("GunStates")
	if mod and mod:IsA("ModuleScript") then
		local ok, r = pcall(require, mod)
		if ok and type(r) == "table" then states = r end
	end
	local orig = Guns.gunOriginals[tool]
	if not orig then
		orig = { vals = {}, attrs = tool:GetAttributes() }
		if states then
			for k, v in pairs(states) do orig.vals[k] = v end
		end
		Guns.gunOriginals[tool] = orig
	end
	local function setVal(key, on, value)
		if states and orig.vals[key] ~= nil then
			states[key] = on and value or orig.vals[key]
		end
		if orig.attrs[key] ~= nil then
			tool:SetAttribute(key, on and value or orig.attrs[key])
		end
	end
	setVal("MaxAmmo", F.gm_ammo, 999999)
	setVal("CurrentAmmo", F.gm_ammo, 999999)
	setVal("AmmoPerClip", F.gm_ammo, 999999)
	setVal("StoredAmmo", F.gm_ammo, 999999)
	setVal("Spread", F.gm_spread, 0)
	setVal("AutoFire", F.gm_auto, true)
	setVal("FireRate", F.gm_rate, F.gm_rate_val or 0.06)
	setVal("Range", F.gm_range, 5000)
	setVal("ReloadTime", F.gm_reload, 0.05)
end

	local c = LP.Character

end

local function espAllowed(p)
	if F.esp_team and p.Team ~= nil and p.Team == LP.Team then return false end
	local tn = teamName(p)
	if tn == "Inmates" then return F.esp_inmates end
	if tn == "Guards" then return F.esp_guards end
	if tn == "Criminals" then return F.esp_criminals end
	return F.esp_neutral
end

local function espColor(p)
	local mode = F.esp_colormode
	if mode == "Theme" then return Theme.accent end
	if mode == "White" then return Color3.new(1, 1, 1) end
	if mode == "Enemy/Ally" then
		return (p.Team ~= nil and p.Team == LP.Team) and NAMED_COLORS.Green or NAMED_COLORS.Red
	end
	return p.Team and p.TeamColor.Color or Color3.fromRGB(200, 200, 200)
end

local function tracerOrigin(vp)
	local mode = F.esp_tracerorigin
	if mode == "Center" then return Vector2.new(vp.X / 2, vp.Y / 2) end
	if mode == "Top" then return Vector2.new(vp.X / 2, 0) end
	if mode == "Mouse" then return UIS:GetMouseLocation() end
	return Vector2.new(vp.X / 2, vp.Y)
end

local function renderEsp()
	local camPos = Camera.CFrame.Position
	local vp = Camera.ViewportSize
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP then
			local o = espCache[p]
			local c = p.Character
			local r = c and c:FindFirstChild("HumanoidRootPart")
			local h = c and c:FindFirstChildOfClass("Humanoid")
			local show = F.esp_enabled and r ~= nil and h ~= nil and h.Health > 0 and espAllowed(p)
			local dist = show and (r.Position - camPos).Magnitude or 0
			if show and dist > F.esp_maxdist then show = false end

			if not show then
				if o then hideEsp(o) end
			else
				if not o then
					o = newEsp()
					espCache[p] = o
				end
				local col = espColor(p)

				if F.esp_chams then
					if not o.hl or o.hl.Adornee ~= c or not o.hl.Parent then
						if o.hl then o.hl:Destroy() end
						o.hl = new("Highlight", { DepthMode = Enum.HighlightDepthMode.AlwaysOnTop, Adornee = c, Parent = overlay })
					end
					o.hl.Enabled = true
					o.hl.FillColor = col
					o.hl.OutlineColor = col
					o.hl.FillTransparency = F.esp_chams_t
				elseif o.hl then
					o.hl.Enabled = false
				end

				local top = Camera:WorldToViewportPoint(r.Position + Vector3.new(0, 3, 0))
				local bot = Camera:WorldToViewportPoint(r.Position - Vector3.new(0, 3.5, 0))
				if top.Z <= 0 or bot.Z <= 0 then
					hide2D(o)
				else
					local hgt = math.max(bot.Y - top.Y, 6)
					local wid = hgt * 0.55
					local x = (top.X + bot.X) / 2
					local y = top.Y

					o.box.Visible = F.esp_boxes
					if F.esp_boxes then
						o.box.Position = UDim2.fromOffset(x - wid / 2, y)
						o.box.Size = UDim2.fromOffset(wid, hgt)
						o.boxStroke.Color = col
					end

					o.name.Visible = F.esp_names
					if F.esp_names then
						o.name.Position = UDim2.fromOffset(x, y - 2)
						o.name.Text = p.DisplayName
						o.name.TextColor3 = col
						o.name.TextSize = F.esp_textsize
					end

					o.info.Visible = F.esp_distance
					if F.esp_distance then
						o.info.Position = UDim2.fromOffset(x, y + hgt + 2)
						o.info.Text = string.format("%dm  |  %s", math.floor(dist), teamName(p))
						o.info.TextSize = math.max(F.esp_textsize - 1, 8)
					end

					o.hpBack.Visible = F.esp_health
					if F.esp_health then
						local hp = math.clamp(h.Health / math.max(h.MaxHealth, 1), 0, 1)
						o.hpBack.Position = UDim2.fromOffset(x - wid / 2 - 6, y)
						o.hpBack.Size = UDim2.fromOffset(3, hgt)
						o.hpFill.Size = UDim2.fromScale(1, hp)
						o.hpFill.BackgroundColor3 = NAMED_COLORS.Red:Lerp(NAMED_COLORS.Green, hp)
					end

					if F.esp_tracers then
						drawLine(o.tracer, tracerOrigin(vp), Vector2.new(x, y + hgt), 1.2)
						o.tracer.BackgroundColor3 = col
					else
						o.tracer.Visible = false
					end
				end
			end
		end
	end
end

local CROSS_STYLES = { "Cross", "Cross + Dot", "Dot", "Circle", "Circle + Dot", "T-Shape", "X" }
local crossBaseRot, crossSpin = 0, 0

local function buildCrosshair()
	crossRoot:ClearAllChildren()
	if not F.ch_enabled then
		crossRoot.Visible = false
		return
	end
	crossRoot.Visible = true
	local col = colorOf(F.ch_color)
	local s, g, t = F.ch_size or 8, F.ch_gap or 4, F.ch_thick or 2
	local style = F.ch_style or "Cross"
	local function bar(w, h, x, y)
		local f = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(w, h), Position = UDim2.fromOffset(x, y),
			BackgroundColor3 = col, Parent = crossRoot,
		})
		if F.ch_outline then stroke(f, Color3.new(0, 0, 0), 1) end
		return f
	end
	local off = g + s / 2
	if style == "Cross" or style == "Cross + Dot" or style == "T-Shape" or style == "X" then
		if style ~= "T-Shape" then bar(t, s, 0, -off) end
		bar(t, s, 0, off)
		bar(s, t, -off, 0)
		bar(s, t, off, 0)
	end
	if style == "Circle" or style == "Circle + Dot" then
		local ring = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(s * 2, s * 2),
			BackgroundTransparency = 1, Parent = crossRoot,
		})
		corner(ring, UDim.new(1, 0))
		stroke(ring, col, t)
		if F.ch_outline then
			local outer = new("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(s * 2 + t * 2, s * 2 + t * 2),
				BackgroundTransparency = 1, Parent = crossRoot,
			})
			corner(outer, UDim.new(1, 0))
			stroke(outer, Color3.new(0, 0, 0), 1, 0.3)
		end
	end
	if style == "Dot" or style == "Cross + Dot" or style == "Circle + Dot" or F.ch_dot then
		local d = bar(t + 2, t + 2, 0, 0)
		corner(d, UDim.new(1, 0))
	end
	crossBaseRot = (style == "X") and 45 or 0
end

	Infin.Setters.ch_gap(gap)
	Infin.Setters.ch_thick(thick)
	if not F.ch_enabled then Infin.Setters.ch_enabled(true) end
end

local function renderOverlay(dt)
	local forced = forcedTarget()
	local fc = forced and forced.Character
	if F.tgt_highlight and fc then
		if targetHighlight.Adornee ~= fc then targetHighlight.Adornee = fc end
		targetHighlight.FillColor = Theme.accent
		targetHighlight.OutlineColor = Theme.accent2
		targetHighlight.Enabled = true
	else
		targetHighlight.Enabled = false
	end

	local origin = aimOrigin()

	local showA = F.aim_enabled and F.aim_showfov
	aimCircle.Visible = showA
	if showA then
		local r = F.aim_fov
		local col = colorOf(F.aim_fovcolor)
		aimCircle.Size = UDim2.fromOffset(r * 2, r * 2)
		aimCircle.Position = UDim2.fromOffset(origin.X, origin.Y)
		aimCircleStroke.Color = col
		aimCircleStroke.Transparency = lockPart and 0 or 0.3
		aimCircle.BackgroundColor3 = col
		aimCircle.BackgroundTransparency = F.aim_fovfill and 0.9 or 1
	end

	local showS = F.silent_enabled and F.silent_showfov
	silentCircle.Visible = showS
	if showS then
		local r = F.silent_fov
		silentCircle.Size = UDim2.fromOffset(r * 2, r * 2)
		silentCircle.Position = UDim2.fromOffset(origin.X, origin.Y)
		silentCircleStroke.Color = colorOf(F.silent_fovcolor)
		silentCircleStroke.Transparency = Infin.SilentPart and 0 or 0.35
	end

	local tPart = Infin.SilentPart or lockPart
	if F.silent_line and tPart then
		local sp, on = Camera:WorldToViewportPoint(tPart.Position)
		if on then
			drawLine(targetLine, origin, Vector2.new(sp.X, sp.Y), 1.5)
			targetLine.BackgroundColor3 = Theme.accent2
		else
			targetLine.Visible = false
		end
	else
		targetLine.Visible = false
	end

	local who = (lockPart and lockTarget) or silentPlayer
	lockLabel.Visible = (who ~= nil) and F.aim_label
	if who then
		lockLabel.Text = (lockPart and "LOCKED  |  " or "TARGET  |  ") .. who.DisplayName
		lockLabel.TextColor3 = Theme.accent
	end

	if F.ch_enabled then
		local vp = Camera.ViewportSize
		local pos = F.ch_follow and UIS:GetMouseLocation() or Vector2.new(vp.X / 2, vp.Y / 2)
		crossRoot.Position = UDim2.fromOffset(pos.X, pos.Y)
		if F.ch_spin then
			crossSpin = (crossSpin + dt * (F.ch_spinspeed or 180)) % 360
		else
			crossSpin = 0
		end
		crossRoot.Rotation = crossBaseRot + crossSpin
	end

	local hideMouse = F.ch_enabled and F.ch_hidemouse and not main.Visible
	if hideMouse then
		UIS.MouseIconEnabled = false
		mouseHiddenByUs = true
	elseif mouseHiddenByUs then
		UIS.MouseIconEnabled = true
		mouseHiddenByUs = false
	end

	if F.fov_custom then Camera.FieldOfView = F.fov_value end
end

bind(RunService.Heartbeat, function()
	if Infin.Unloaded then return end
	local h, r = getHum(), getRoot()
	if h then
		if F.mv_speed then
			h.WalkSpeed = F.mv_speed_val
		elseif F.mv_antitase and h.WalkSpeed < 16 then
			h.WalkSpeed = 16
		end
		if F.mv_jump then
			h.UseJumpPower = true
			h.JumpPower = F.mv_jump_val
		elseif F.mv_antitase and h.UseJumpPower and h.JumpPower < 50 then
			h.JumpPower = 50
		end
	end
	if F.mv_fly and h and r then
		local speed = F.mv_fly_val
		local md = h.MoveDirection
		local look = Camera.CFrame.LookVector
		local vel = md * speed
		if md.Magnitude > 0 then
			local flat = Vector3.new(look.X, 0, look.Z)
			if flat.Magnitude > 0 then
				vel += Vector3.new(0, look.Y * speed * md:Dot(flat.Unit), 0)
			end
		end
		if UIS:IsKeyDown(Enum.KeyCode.Space) then
			vel += Vector3.new(0, speed, 0)
		elseif UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
			vel -= Vector3.new(0, speed, 0)
		end
		r.AssemblyLinearVelocity = vel
	end
	if F._looptp then
		local p = selected()
		local tr = p and getRoot(p)
		if tr and r then r.CFrame = tr.CFrame * CFrame.new(0, 0, 3) end
	end
end)

local lightingOrig = {}
local function storeLighting(keys)
	for _, k in ipairs(keys) do
		if lightingOrig[k] == nil then lightingOrig[k] = Lighting[k] end
	end
end
local function restoreLighting(keys)
	for _, k in ipairs(keys) do
		if lightingOrig[k] ~= nil then
			pcall(function() Lighting[k] = lightingOrig[k] end)
			lightingOrig[k] = nil
		end
	end
end

local function serverHop()
	local ok, body = pcall(function()
		return game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Desc&limit=100")
	end)
	if not ok or not body then notify("Server list unavailable") return end
	local okJ, data = pcall(function() return HttpService:JSONDecode(body) end)
	if not okJ or type(data) ~= "table" then notify("Server list unavailable") return end
	local options = {}
	for _, s in ipairs(data.data or {}) do
		if s.id ~= game.JobId and (s.playing or 0) < (s.maxPlayers or 0) then table.insert(options, s.id) end
	end
	if #options == 0 then notify("No other servers found") return end
	notify("Hopping servers...")
	TeleportService:TeleportToPlaceInstance(game.PlaceId, options[math.random(#options)], LP)
end

	local h = c:WaitForChild("Humanoid", 8)

		end)

		teleportTo(pendingReturn.cf)
		pendingReturn = nil
	elseif F.mv_return and lastDeathCF then

		task.wait(0.6)
		if F.wp_autoitem == "All Guns" then
			Guns.giveItems(GUNS)
		else

	end

local Config = {}

function Config.fsOK()
	return writefile and readfile and isfile and makefolder and isfolder and true or false
end
function Config.ensureFolder()
	if not isfolder(CONFIG_FOLDER) then makefolder(CONFIG_FOLDER) end
end
function Config.configPath(slot) return CONFIG_FOLDER .. "/config" .. tostring(slot) .. ".json" end
Config.AUTOLOAD_PATH = CONFIG_FOLDER .. "/autoload.txt"

function Config.saveConfig(slot)
	if not Config.fsOK() then notify("Executor has no file access") return end
	Config.ensureFolder()
	local data = {}
	for k, v in pairs(F) do
		if string.sub(k, 1, 1) ~= "_" then data[k] = v end
	end
	writefile(Config.configPath(slot), HttpService:JSONEncode(data))
	notify("Saved config " .. tostring(slot))
end

function Config.loadConfig(slot, quiet)
	if not Config.fsOK() then
		if not quiet then notify("Executor has no file access") end
		return
	end
	if not isfile(Config.configPath(slot)) then
		if not quiet then notify("Config " .. tostring(slot) .. " is empty") end
		return
	end
	local ok, data = pcall(function() return HttpService:JSONDecode(readfile(Config.configPath(slot))) end)
	if not ok or type(data) ~= "table" then notify("Config " .. tostring(slot) .. " is corrupted") return end
	for k, v in pairs(data) do
		local s = Infin.Setters[k]
		if s and k ~= "cfg_slot" and k ~= "cfg_autoload" then safe("load " .. k, s, v) end
	end
	if not quiet then notify("Loaded config " .. tostring(slot)) end
end

function Config.resetConfig()
	for k, def in pairs(Infin.Defaults) do
		if k ~= "cfg_slot" and k ~= "cfg_autoload" then safe("reset " .. k, Infin.Setters[k], def) end
	end
	notify("Reset to defaults")
end

function Config.setAutoload(on)
	if not Config.fsOK() then return end
	Config.ensureFolder()
	if on then
		writefile(Config.AUTOLOAD_PATH, tostring(F.cfg_slot or "1"))
	elseif isfile(Config.AUTOLOAD_PATH) and delfile then
		delfile(Config.AUTOLOAD_PATH)
	end
end

local autoloadSlot
if Config.fsOK() then
	pcall(function()
		if isfile(Config.AUTOLOAD_PATH) then autoloadSlot = readfile(Config.AUTOLOAD_PATH) end
	end)
end

local function unload()
	if Infin.Unloaded then return end
	for flag, def in pairs(Infin.Defaults) do
		if type(def) == "boolean" and F[flag] and flag ~= "cfg_autoload" and flag ~= "ui_notify" then
			pcall(Infin.Setters[flag], false)
		end
	end
	Infin.Unloaded = true
	Infin.SilentPos, Infin.SilentPart = nil, nil
	for _, c in ipairs(Infin.Conns) do pcall(function() c:Disconnect() end) end
	pcall(function() RunService:UnbindFromRenderStep("InfinPrisonRender") end)
	for p in pairs(espCache) do removeEsp(p) end
	setDoors(false)
	restoreLighting(FB_KEYS)
	restoreLighting(FOG_KEYS)
	if mouseHiddenByUs then UIS.MouseIconEnabled = true end
	local h = getHum()
	if h then Camera.CameraSubject = h end
	pcall(function() gui:Destroy() end)
	pcall(function() overlay:Destroy() end)
	genv.INFIN_PRISON_UNLOAD = nil
end

local fxTime = 0
bind(RunService.Heartbeat, function(dt)
	if Infin.Unloaded or not main.Visible then return end
	fxTime += dt
	borderGrad.Rotation = (fxTime * 60) % 360
	if F.fx_stars then
		for _, s in ipairs(stars) do
			local wave = 0.5 + 0.5 * math.sin(fxTime * s.speed + s.phase)
			s.f.BackgroundTransparency = s.low + (0.85 - s.low) * wave
		end
	end
	if F.fx_dots then
		for _, d in ipairs(dots) do
			d.x = (d.x + d.vx * dt) % 1
			d.y = (d.y + d.vy * dt) % 1
			d.f.Position = UDim2.fromScale(d.x, d.y)
		end
	end
end)

task.spawn(function()
	while not Infin.Unloaded do
		task.wait(1)
		fpsLabel.Text = "FPS " .. frameCount
		frameCount = 0
		local ping
		pcall(function() ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
		if not ping then pcall(function() ping = math.floor(LP:GetNetworkPing() * 2000) end) end
		if ping then
			pingLabel.Text = "PING " .. ping .. "ms"
			pingLabel.TextColor3 = (ping < 90 and NAMED_COLORS.Green) or (ping < 160 and NAMED_COLORS.Yellow) or NAMED_COLORS.Red
		end
		if setTeamInfo then
			local counts = { Inmates = 0, Guards = 0, Criminals = 0, Neutral = 0 }
			for _, p in ipairs(Players:GetPlayers()) do
				local tn = teamName(p)
				counts[tn] = (counts[tn] or 0) + 1
			end
			setTeamInfo(string.format(
				'<font color="#FF9A3C">Inmates</font>  %d\n<font color="#4C8DFF">Guards</font>  %d\n<font color="#FF4B5C">Criminals</font>  %d\nNeutral  %d\n\nYou: <b>%s</b>',
				counts.Inmates, counts.Guards, counts.Criminals, counts.Neutral, teamName(LP)
			))
		end
	end
end)
