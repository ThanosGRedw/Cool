local function updateVelocity(productNumber)
	local updateVelocityState = nil
	local secondaryUpdateVelocityState = nil
	local product = nil
	local updateCFrameNumber = nil
	local alternateUpdateVelocityState = nil
	local additionalUpdateVelocityState = nil
	local fallbackUpdateVelocityState = nil
	local nestedUpdateVelocityState = nil
	local innerUpdateVelocityState = nil
	local outerUpdateVelocityState = nil
	local previousUpdateVelocityState = nil
	local currentUpdateVelocityState = nil
	local nextUpdateVelocityState = nil
	local sourceUpdateVelocityState = nil
	local targetUpdateVelocityState = nil

	while 0 ~= 0 do
	end

	local Character = LocalPlayer.Character

	if Character and Character:FindFirstChild("Humanoid", updateVelocityState, secondaryUpdateVelocityState) and Character:FindFirstChild("HumanoidRootPart", updateVelocityState, secondaryUpdateVelocityState) and updateVelocityFlag then
		local number = 0
		local Humanoid = nil
		local HumanoidRootPart = nil

		while true do
			if number == 0 then
				Humanoid = Character.Humanoid
				HumanoidRootPart = Character.HumanoidRootPart
				number = 1
			end

			if number == 3 then
				break
			end

			if number == 2 then
				local MoveDirection = Humanoid.MoveDirection
				local PivotTo = Character.PivotTo
				local Pivot = Character:GetPivot()
				local secondaryUpdateVelocityNumber = MoveDirection * productNumber

				updateCFrameNumber = updateVelocityNumber
				product = secondaryUpdateVelocityNumber * updateCFrameNumber
				PivotTo(Character, Pivot + product, product, updateCFrameNumber, alternateUpdateVelocityState, additionalUpdateVelocityState, fallbackUpdateVelocityState, nestedUpdateVelocityState)
				number = 3
			end

			if number == 1 then
				if Humanoid.Sit == false then
					Humanoid:ChangeState(Enum.HumanoidStateType.Climbing, product, updateCFrameNumber, alternateUpdateVelocityState, additionalUpdateVelocityState, fallbackUpdateVelocityState, nestedUpdateVelocityState)
				end

				HumanoidRootPart.Velocity = Vector3.zero
				number = 2
			end
		end

		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift, product, updateCFrameNumber, alternateUpdateVelocityState, additionalUpdateVelocityState, fallbackUpdateVelocityState, nestedUpdateVelocityState) then
			local PivotTo = Character.PivotTo
			local Pivot = Character:GetPivot()
			local new = Vector3.new
			local newNumber = 0
			local product = updateVelocityNumber * productNumber
			local number = 0
			local secondaryUpdateVelocityNumber = new(newNumber, product, number, fallbackUpdateVelocityState, nestedUpdateVelocityState, innerUpdateVelocityState, outerUpdateVelocityState, previousUpdateVelocityState, currentUpdateVelocityState, nextUpdateVelocityState, sourceUpdateVelocityState, targetUpdateVelocityState)

			PivotTo(Character, Pivot - secondaryUpdateVelocityNumber, secondaryUpdateVelocityNumber, newNumber, product, number, fallbackUpdateVelocityState, nestedUpdateVelocityState)
		end
	end
end

	local nestedUpdateCFrameState = nil

	local outerUpdateCFrameState = nil
	local previousUpdateCFrameState = nil
	local currentUpdateCFrameState = nil
	local nextUpdateCFrameState = nil
	local updateCFrameNumber = 0
	local Character = nil

	while updateCFrameNumber ~= 1 do
		if updateCFrameNumber == 0 then
			if not capturedUpdateCFrameFlag then
				return
			end

			Character = LocalPlayer.Character
			updateCFrameNumber = 1
		end
	end

	if Character and Character:FindFirstChild("Humanoid", updateCFrameState, secondaryUpdateCFrameState) and Character:FindFirstChild("HumanoidRootPart", updateCFrameState, secondaryUpdateCFrameState) then
		local updateCFrameNumber = 0
		local Humanoid = nil
		local HumanoidRootPart = nil
		local MoveDirection = nil

		repeat
			if updateCFrameNumber == 1 then
				Humanoid:ChangeState(Enum.HumanoidStateType.Swimming, alternateUpdateCFrameState, additionalUpdateCFrameState, fallbackUpdateCFrameState, nestedUpdateCFrameState, innerUpdateCFrameState, outerUpdateCFrameState)
				MoveDirection = Humanoid.MoveDirection
				updateCFrameNumber = 2
			end

			if updateCFrameNumber == 0 then
				Humanoid = Character.Humanoid
				HumanoidRootPart = Character.HumanoidRootPart
				updateCFrameNumber = 1
			end
		until updateCFrameNumber == 2

		HumanoidRootPart.CFrame = HumanoidRootPart.CFrame + MoveDirection * updateVelocityNumber * number

		if not UserInputService:IsKeyDown(Enum.KeyCode.Space, alternateUpdateCFrameState, additionalUpdateCFrameState, fallbackUpdateCFrameState, nestedUpdateCFrameState, innerUpdateCFrameState, outerUpdateCFrameState) then
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift, alternateUpdateCFrameState, additionalUpdateCFrameState, fallbackUpdateCFrameState, nestedUpdateCFrameState, innerUpdateCFrameState, outerUpdateCFrameState) then
				HumanoidRootPart.CFrame = HumanoidRootPart.CFrame - Vector3.new(0, cFrameNumber * number, 0, fallbackUpdateCFrameState, nestedUpdateCFrameState, innerUpdateCFrameState, outerUpdateCFrameState, previousUpdateCFrameState, currentUpdateCFrameState, nextUpdateCFrameState)
			end
		else
			HumanoidRootPart.CFrame = HumanoidRootPart.CFrame + Vector3.new(0, cFrameNumber * number, 0, fallbackUpdateCFrameState, nestedUpdateCFrameState, innerUpdateCFrameState, outerUpdateCFrameState, previousUpdateCFrameState, currentUpdateCFrameState, nextUpdateCFrameState)
		end
	end
end
