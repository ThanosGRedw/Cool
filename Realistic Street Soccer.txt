-- Lógica do Aimbot Ball integrando detecção estrita de bola
local bolaCache = nil
local function obterBolaReal()
    if bolaCache and bolaCache:IsDescendantOf(i) and bolaCache:IsA("BasePart") and not bolaCache.Anchored then
        return bolaCache
    end
    
    local possivelBola = i:FindFirstChild("TPS") or i:FindFirstChild("Ball") or i:FindFirstChild("Football")
    if possivelBola then
        if possivelBola:IsA("BasePart") and not possivelBola.Anchored then
            bolaCache = possivelBola
            return possivelBola
        elseif possivelBola:IsA("Model") and possivelBola.PrimaryPart and not possivelBola.PrimaryPart.Anchored then
            bolaCache = possivelBola.PrimaryPart
            return possivelBola.PrimaryPart
        end
    end

    for _, obj in ipairs(i:GetChildren()) do
        local nome = obj.Name:lower()
        if (nome == "ball" or nome == "tps" or nome == "football" or nome == "bola" or nome:find("matchball")) then
            if obj:IsA("BasePart") and not obj.Anchored and obj.CanCollide then
                bolaCache = obj
                return obj
            elseif obj:IsA("Model") then
                local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                if part and not part.Anchored then
                    bolaCache = part
                    return part
                end
            end
        end
    end
    
    bolaCache = nil
    return nil
end

local function executarAimbotBall()
    local char = k.Character
    if not char then return end
    
    local hrp = char:FindFirstChild(a[11])
    if not hrp then return end
    
    local bola = obterBolaReal()
    if not bola then return end
    
    local posBola = bola.Position
    local lookAtPos = Vector3.new(posBola.X, hrp.Position.Y, posBola.Z)
    hrp.CFrame = CFrame.new(hrp.Position, lookAtPos)
end
